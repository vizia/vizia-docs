18///Creating a new application creates a root `Window` and a `Context`. Views declared within the closure passed to `Application::new()` are added to the context and rendered into the root window.
19///
20/// # Example
21/// ```no_run
22/// # use vizia_core::prelude::*;
23/// # use vizia_baseview::Application;
24///
25/// Application::new(|cx|{
26///    // Content goes here
27/// })
28/// .run();
29///```
30/// Calling `run()` on the `Application` causes the program to enter the event loop and for the main window to display.
31pub struct Application<F>
32where
33    F: Fn(&mut Context) + Send + 'static,
34{
35    app: F,
36    window_description: WindowDescription,
37    window_scale_policy: WindowScalePolicy,
38    frame_policy: FramePolicy,
39    on_idle: Option<Box<dyn Fn(&mut Context) + Send>>,
40    ignore_default_theme: bool,
41}

147    /// Takes a closure which will be called at the end of every loop of the application.
148    ///
149    /// The callback provides a place to run 'idle' processing and happens at the end of each loop but before drawing.
150    /// If the callback pushes events into the queue in context then the event loop will re-run. Care must be taken not to
151    /// push events into the queue every time the callback runs unless this is intended.
152    ///
153    /// # Example
154    /// ```no_run
155    /// # use vizia_core::prelude::*;
156    /// # use vizia_baseview::Application;
157    /// Application::new(|cx|{
158    ///     // Build application here
159    /// })
160    /// .on_idle(|cx|{
161    ///     // Code here runs at the end of every event loop after OS and vizia events have been handled
162    /// })
163    /// .run();
164    /// ```
165    pub fn on_idle<I: 'static + Fn(&mut Context) + Send>(mut self, callback: I) -> Self {
166        self.on_idle = Some(Box::new(callback));
167
168        self
169    }

238    /// Apply a new user scale factor: resize the embedded window,
239    /// update vizia's tracked DPI factor + root viewport bounds, and
240    /// recreate the Skia render surfaces at the new physical size.
241    ///
242    /// Driven by `WindowEvent::SetUserScale(f64)`. baseview does not
243    /// emit a `Resized` event in response to a self-driven
244    /// `Window::resize` (the only Resized trigger on macOS is
245    /// `viewDidChangeBackingProperties`, which fires on backing-scale
246    /// changes only) so this function performs every step the
247    /// `Resized` handler does, plus the `Window::resize` call.
248    fn apply_user_scale(&mut self, new_user_scale: f64) {
249        self.window_description.user_scale_factor = new_user_scale;
250
251        // baseview treats the user scale factor as part of its
252        // logical size, so the size we hand to `Window::resize` is
253        // already user-scaled. The HiDPI factor is applied on top by
254        // baseview itself when computing pixel dims.
255        let base_logical_w = self.window_description.inner_size.width as f64;
256        let base_logical_h = self.window_description.inner_size.height as f64;
257        let scaled_logical_w = base_logical_w * new_user_scale;
258        let scaled_logical_h = base_logical_h * new_user_scale;
259        self.window_context
260            .resize(baseview::dpi::LogicalSize::new(scaled_logical_w, scaled_logical_h))
261            .expect("Failed to resize window");
262
263        let new_dpi_factor = self.window_scale_factor * new_user_scale;
264        self.cx.set_scale_factor(new_dpi_factor);
265
266        let new_physical_w = (scaled_logical_w * self.window_scale_factor) as f32;
267        let new_physical_h = (scaled_logical_h * self.window_scale_factor) as f32;
268        self.cx.set_window_size(Entity::root(), new_physical_w, new_physical_h);
269
270        if new_physical_w > 0.0 && new_physical_h > 0.0 {
271            let fb_info = {
272                let mut fboid: GLint = 0;
273                unsafe { gl::GetIntegerv(gl::FRAMEBUFFER_BINDING, &mut fboid) };
274
275                FramebufferInfo {
276                    fboid: fboid.try_into().unwrap(),
277                    format: skia_safe::gpu::gl::Format::RGBA8.into(),
278                    ..Default::default()
279                }
280            };
281
282            self.surface = create_surface(
283                (new_physical_w as i32, new_physical_h as i32),
284                fb_info,
285                &mut self.gr_context,
286            );
287
288            self.dirty_surface = self
289                .surface
290                .new_surface_with_dimensions((
291                    new_physical_w.max(1.0) as i32,
292                    new_physical_h.max(1.0) as i32,
293                ))
294                .unwrap();
295        }
296
297        self.cx.needs_refresh(Entity::root());
298    }

302    pub fn on_frame_update(&mut self) -> bool {
303        // Pick up any effects enqueued by off-UI-thread `SyncSignal` writes since the last
304        // frame. These sit in `SYNC_RUNTIME` until a UI-thread call processes them — this
305        // is the analogue of the `drain_pending_work` call in `vizia_winit`'s frame loop.
306        Runtime::drain_pending_work();
307
308        for event in self.proxy_queue.take_events() {
309            self.cx.send_event(event.into_event());
310        }
311
312        // Fire due timers and scheduled events before flushing so they're handled this
313        // frame, mirroring `vizia_winit`'s `new_events`.
314        self.cx.process_timers();
315        self.cx.emit_scheduled_events();
316
317        // Events. The flush callback can't borrow `&mut self`, so size /
318        // scale change requests get latched into locals here and applied
319        // after the drain.
320        let mut pending_user_scale: Option<f64> = None;
321        self.event_manager.flush_events(self.cx.context(), |window_event| {
322            // For some reason calling window.close() crashes baseview on macos
323            // WindowEvent::WindowClose => *should_close = true,
324            match window_event {
325                WindowEvent::FocusIn => {
326                    #[cfg(not(target_os = "linux"))] // not implemented for linux yet
327                    if !self.window_context.has_focus() {
328                        self.window_context.focus().expect("Failed to focus window");
329                    }
330                }
331                WindowEvent::SetUserScale(factor) => {
332                    pending_user_scale = Some(*factor);
333                }
334                WindowEvent::SetCursor(icon) => {
335                    _ = self.window_context.set_mouse_cursor(translate_mouse_cursor(*icon))
336                }
337                _ => {}
338            }
339        });
340
341        if let Some(new_user_scale) = pending_user_scale {
342            self.apply_user_scale(new_user_scale);
343        }
344
345        // We need to resize the window to make sure that the new size is applied. This is a workaround
346        // for the fact that baseview does not resize the window when the scale factor changes.
347        if !self.is_initialized {
348            // Resizing the window doesn't apply unless the size has actually changed.
349            // So we resize the window slightly larger and then back again to force a resize event.

355                .expect("Failed to resize window");
356

358                .resize(baseview::dpi::LogicalSize::new(
359                    self.window_description.inner_size.width as f64,

454    pub fn render(&mut self) {
455        if self.should_redraw {
456            let context = self
457                .window_context
458                .gl_context()
459                .expect("Window was created without OpenGL support");
460            unsafe { context.make_current() }.expect("Failed to make GL context current");
461            self.cx.draw(Entity::root(), &mut self.surface, &mut self.dirty_surface);
462            self.gr_context.flush_and_submit();
463            self.should_redraw = false;
464            context.swap_buffers().expect("Failed to swap buffers");
465            unsafe { context.make_not_current() }.expect("Failed to make GL context not current");
466        }
467    }

469    pub fn handle_event(&mut self, event: baseview::Event) -> baseview::EventStatus {
470        let status = self.event_status(&event);
471
472        if requests_exit(&event, self.is_parented) {
473            self.cx.send_event(Event::new(WindowEvent::WindowClose));
474            self.window_context.request_close();
475        }
476
477        let mut update_modifiers = |modifiers: vizia_input::KeyboardModifiers| {
478            self.cx
479                .modifiers()
480                .set(Modifiers::SHIFT, modifiers.contains(vizia_input::KeyboardModifiers::SHIFT));
481            self.cx
482                .modifiers()
483                .set(Modifiers::CTRL, modifiers.contains(vizia_input::KeyboardModifiers::CONTROL));
484            self.cx
485                .modifiers()
486                .set(Modifiers::SUPER, modifiers.contains(vizia_input::KeyboardModifiers::META));
487            self.cx
488                .modifiers()
489                .set(Modifiers::ALT, modifiers.contains(vizia_input::KeyboardModifiers::ALT));
490        };
491
492        match event {
493            baseview::Event::Mouse(event) => match event {
494                baseview::MouseEvent::CursorMoved { position, modifiers } => {
495                    update_modifiers(modifiers);
496
497                    // baseview v0.4 delivers cursor coordinates in physical pixels, so no
498                    // additional DPI scaling should be applied here.
499                    let cursor_x = position.x as f32;
500                    let cursor_y = position.y as f32;
501                    self.cx.emit_origin(WindowEvent::MouseMove(cursor_x, cursor_y));
502                }
503                baseview::MouseEvent::ButtonPressed { button, modifiers } => {
504                    update_modifiers(modifiers);
505
506                    let b = translate_mouse_button(button);
507                    self.cx.emit_origin(WindowEvent::MouseDown(b));
508                }
509                baseview::MouseEvent::ButtonReleased { button, modifiers } => {
510                    update_modifiers(modifiers);
511
512                    let b = translate_mouse_button(button);
513                    self.cx.emit_origin(WindowEvent::MouseUp(b));
514                }
515                baseview::MouseEvent::WheelScrolled { delta, modifiers } => {
516                    update_modifiers(modifiers);
517
518                    let (lines_x, lines_y) = match delta {
519                        baseview::ScrollDelta::Lines { x, y } => (x, y),
520                        baseview::ScrollDelta::Pixels { x, y } => (
521                            if x < 0.0 {
522                                -1.0
523                            } else if x > 1.0 {
524                                1.0
525                            } else {
526                                0.0
527                            },
528                            if y < 0.0 {
529                                -1.0
530                            } else if y > 1.0 {
531                                1.0
532                            } else {
533                                0.0
534                            },
535                        ),
536                    };
537
538                    self.cx.emit_origin(WindowEvent::MouseScroll(lines_x, lines_y));
539                }
540
541                baseview::MouseEvent::CursorEntered => {
542                    self.cx.emit_origin(WindowEvent::MouseEnter);
543                }
544
545                baseview::MouseEvent::CursorLeft => {
546                    self.cx.emit_origin(WindowEvent::MouseLeave);
547                }
548
549                baseview::MouseEvent::DragDropped { modifiers, data, .. } => {
550                    update_modifiers(modifiers);
551
552                    if let baseview::DropData::Files(files) = data {
553                        for file in files {
554                            // Try mimic vizia winit drop handling
555                            self.cx.emit_window_event(
556                                Entity::root(),
557                                WindowEvent::Drop(DropData::File(file)),
558                            );
559                        }
560                    }
561                }
562
563                // Vizia winit discard HoveredFile | HoveredFileCancelled
564                // so we mirror the behavior on baseview.
565                baseview::MouseEvent::DragEntered { .. }
566                | baseview::MouseEvent::DragLeft
567                | baseview::MouseEvent::DragMoved { .. } => {}
568
569                _ => {}
570            },
571            baseview::Event::Keyboard(event) => {
572                let (s, pressed) = match event.state {
573                    vizia_input::KeyState::Down => (MouseButtonState::Pressed, true),
574                    vizia_input::KeyState::Up => (MouseButtonState::Released, false),
575                };
576
577                match event.code {
578                    Code::ShiftLeft | Code::ShiftRight => {
579                        self.cx.modifiers().set(Modifiers::SHIFT, pressed)
580                    }
581                    Code::ControlLeft | Code::ControlRight => {
582                        self.cx.modifiers().set(Modifiers::CTRL, pressed)
583                    }
584                    Code::AltLeft | Code::AltRight => {
585                        self.cx.modifiers().set(Modifiers::ALT, pressed)
586                    }
587                    Code::MetaLeft | Code::MetaRight => {
588                        self.cx.modifiers().set(Modifiers::SUPER, pressed)
589                    }
590                    _ => (),
591                }
592
593                match s {
594                    MouseButtonState::Pressed => {
595                        if let vizia_input::Key::Character(written) = &event.key {
596                            for chr in written.chars() {
597                                self.cx.emit_origin(WindowEvent::CharInput(chr));
598                            }
599                        }
600
601                        self.cx.emit_origin(WindowEvent::KeyDown(event.code, Some(event.key)));
602                    }
603
604                    MouseButtonState::Released => {
605                        self.cx.emit_origin(WindowEvent::KeyUp(event.code, Some(event.key)));
606                    }
607                }
608            }
609            baseview::Event::Window(event) => match event {
610                baseview::WindowEvent::Focused => self.cx.needs_refresh(Entity::root()),
611                baseview::WindowEvent::WillClose => {
612                    self.cx.send_event(Event::new(WindowEvent::WindowClose));
613                }
614                _ => {}
615            },
616            _ => {}
617        }
618
619        status
620    }

622    pub fn handle_resized(&mut self, new_size: baseview::WindowSize) {
623        let context =
624            self.window_context.gl_context().expect("Window was created without OpenGL support");
625        unsafe { context.make_current() }.expect("Failed to make GL context current");
626
627        let fb_info = {
628            let mut fboid: GLint = 0;
629            unsafe { gl::GetIntegerv(gl::FRAMEBUFFER_BINDING, &mut fboid) };
630
631            FramebufferInfo {
632                fboid: fboid.try_into().unwrap(),
633                format: skia_safe::gpu::gl::Format::RGBA8.into(),
634                ..Default::default()
635            }
636        };
637
638        let width = new_size.physical.width.max(1) as i32;
639        let height = new_size.physical.height.max(1) as i32;
640
641        self.surface = create_surface((width, height), fb_info, &mut self.gr_context);
642
643        self.dirty_surface = self.surface.new_surface_with_dimensions((width, height)).unwrap();
644
645        unsafe { context.make_not_current() }.expect("Failed to make GL context not current");
646
647        self.window_scale_factor = new_size.scale_factor;
648
649        self.cx
650            .set_scale_factor(self.window_scale_factor * self.window_description.user_scale_factor);
651
652        self.cx.set_window_size(
653            Entity::root(),
654            new_size.physical.width as f32,
655            new_size.physical.height as f32,
656        );
657
658        self.cx.needs_refresh(Entity::root());
659    }

661    pub fn handle_idle(&mut self, on_idle: &Option<Box<dyn Fn(&mut Context) + Send>>) {
662        if let Some(idle_callback) = on_idle {
663            self.cx.set_current(Entity::root());
664            (idle_callback)(self.cx.context());
665        }
666    }

699    /// Returns the [`EventStatus`](`baseview::EventStatus`) for this `event`
700    fn event_status(&self, event: &baseview::Event) -> baseview::EventStatus {
701        match event {
702            // macOS and Windows only perform the drop if
703            // the drag-enter/move callbacks report an acceptable operation.
704            // The events can still be ignored inside `handle_event`.
705            // (X11 ignores the return status.)
706            baseview::Event::Mouse(
707                baseview::MouseEvent::DragEntered { .. }
708                | baseview::MouseEvent::DragMoved { .. }
709                | baseview::MouseEvent::DragDropped { .. },
710            ) => baseview::EventStatus::AcceptDrop(baseview::DropEffect::Copy),
711
712            // If a text input currently holds focus, a native keyboard event is
713            // intended for the plugin. Report it as `Captured` so the backend
714            // does not forward the platform event further — otherwise the host
715            // (e.g. a DAW) also processes the same keystroke, producing double
716            // handling and a system beep on macOS.
717            baseview::Event::Keyboard(_) if self.focused_element() == Some("textbox") => {
718                baseview::EventStatus::Captured
719            }
720
721            _ => baseview::EventStatus::Ignored,
722        }
723    }

726/// Returns true if the provided event should cause an [`Application`] to
727/// exit.
728///
729/// `WindowEvent::WillClose` is honoured in both standalone and parented
730/// modes — it's a legitimate close signal from baseview / the host.
731///
732/// On macOS, Cmd+Q is recognized as a quit shortcut **only when the
733/// application is standalone** (`is_parented == false`). When
734/// vizia_baseview is embedded as a child window inside a host (the
735/// usual audio-plug-in setup), the host owns the application
736/// lifecycle: pressing Cmd+Q should quit the host, not close the
737/// plug-in's child window. Most hosts on macOS bind Cmd+Q to their
738/// own menu's "Quit" item, so AppKit's `performKeyEquivalent:`
739/// dispatch claims the key before the plug-in's NSView sees it.
740/// Hosts with looser key dispatch (Bitwig observed 2026-04-28) do
741/// forward it through, in which case the previous unconditional
742/// match would tear down the plug-in's GL surface and leave the
743/// host with an empty plug-in shell.
744pub fn requests_exit(
745    event: &baseview::Event,
746    #[cfg_attr(not(target_os = "macos"), allow(unused_variables))] is_parented: bool,
747) -> bool {
748    match event {
749        baseview::Event::Window(baseview::WindowEvent::WillClose) => true,
750        #[cfg(target_os = "macos")]
751        baseview::Event::Keyboard(event) if !is_parented => {
752            if event.code == vizia_input::Code::KeyQ
753                && event.modifiers == vizia_input::KeyboardModifiers::META
754                && event.state == vizia_input::KeyState::Down
755            {
756                return true;
757            }
758
759            false
760        }
761        _ => false,
762    }
763}
