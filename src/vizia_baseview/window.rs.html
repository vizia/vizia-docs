25    #[allow(clippy::type_complexity)]

31impl ViziaWindow {
32    fn new(
33        mut cx: BackendContext,
34        win_desc: WindowDescription,
35        frame_policy: FramePolicy,
36        window: WindowContext,
37        builder: Option<Box<dyn FnOnce(&mut Context) + Send>>,
38        on_idle: Option<Box<dyn Fn(&mut Context) + Send>>,
39        is_parented: bool,
40    ) -> ViziaWindow {
41        // Reactive runtime setup — mirrors what `vizia_winit::Application::new` does.
42        //
43        // Mark this thread as the UI thread so off-thread `SyncSignal` writes correctly
44        // enqueue effects via `SYNC_RUNTIME` instead of the thread-local `RUNTIME` (which
45        // nobody drains). `on_frame_update` later calls `Runtime::drain_pending_work` to
46        // apply those effects on this thread.
47        Runtime::init_on_ui_thread();
48
49        // The runner owns the queue and drains it in `on_frame_update`.
50        // proxies only hold a weak reference, so sending fails once the window is closed.
51        let proxy_queue = ProxyQueue::default();
52        cx.set_event_proxy(Box::new(proxy_queue.proxy(window.waker())));
53
54        // Register the window as needing wake only if the policy is `OnDemand`.
55        // `Continuous` draw runs every vsync frame anyway.
56        let guard =
57            (frame_policy == FramePolicy::OnDemand).then(|| register_window(window.waker()));
58        Runtime::set_sync_effect_waker(wake);
59
60        let context = window.gl_context().expect("Window was created without OpenGL support");
61
62        unsafe { context.make_current() }.expect("Failed to make GL context current");
63
64        // Build skia renderer
65        gl::load_with(|s| context.get_proc_address_from_str(s));
66        let interface = skia_safe::gpu::gl::Interface::new_load_with(|name| {
67            if name == "eglGetCurrentDisplay" {
68                return std::ptr::null();
69            }
70            context.get_proc_address_from_str(name)
71        })
72        .expect("Could not create interface");
73
74        // https://github.com/rust-skia/rust-skia/issues/476
75        let mut context_options = ContextOptions::new();
76        context_options.skip_gl_error_checks = context_options::Enable::Yes;
77
78        let mut gr_context = skia_safe::gpu::direct_contexts::make_gl(interface, &context_options)
79            .expect("Could not create direct context");
80        gr_context.set_resource_cache_limit(win_desc.skia_resource_cache_limit);
81
82        let fb_info = {
83            let mut fboid: GLint = 0;
84            unsafe { gl::GetIntegerv(gl::FRAMEBUFFER_BINDING, &mut fboid) };
85
86            FramebufferInfo {
87                fboid: fboid.try_into().unwrap(),
88                format: skia_safe::gpu::gl::Format::RGBA8.into(),
89                ..Default::default()
90            }
91        };
92
93        let initial_size = window.size();
94        let initial_width = initial_size.physical.width.max(1) as i32;
95        let initial_height = initial_size.physical.height.max(1) as i32;
96
97        let mut surface = create_surface((initial_width, initial_height), fb_info, &mut gr_context);
98
99        let dirty_surface =
100            surface.new_surface_with_dimensions((initial_width, initial_height)).unwrap();
101
102        // baseview determines the scale factor from the platform. The
103        // `WindowScalePolicy::ScaleFactor(scale)`
104        // value is only used as a fallback (via
105        // `with_fallback_scale_factor` in `open_parented`/`open_blocking`) when the platform
106        // cannot report one, so always read the actual scale factor here to keep vizia's layout consistent with the
107        // real window.
108        let window_scale_factor = window.scale_factor();
109        let dpi_factor = window_scale_factor * win_desc.user_scale_factor;
110
111        cx.add_main_window(Entity::root(), &win_desc, dpi_factor as f32);
112        cx.add_window(Entity::root(), WindowView {});
113
114        cx.0.windows.insert(
115            Entity::root(),
116            WindowState { window_description: win_desc.clone(), ..Default::default() },
117        );
118
119        cx.context().add_built_in_styles();
120        if let Some(builder) = builder {
121            (builder)(cx.context());
122        }
123
124        let application = ApplicationRunner::new(
125            cx,
126            gr_context,
127            window_scale_factor,
128            frame_policy,
129            proxy_queue,
130            surface,
131            dirty_surface,
132            win_desc,
133            is_parented,
134            window,
135        );
136        unsafe { context.make_not_current() }.expect("Failed to make GL context not current");
137
138        ViziaWindow { application: RefCell::new(application), on_idle, _sync_waker: guard }
139    }

141    /// Open a new child window.
142    ///
143    /// * `parent` - The parent window.
144    /// * `app` - The Vizia application builder.
145    pub fn open_parented<P, F>(
146        parent: &P,
147        win_desc: WindowDescription,
148        scale_policy: WindowScalePolicy,
149        frame_policy: FramePolicy,
150        app: F,
151        on_idle: Option<Box<dyn Fn(&mut Context) + Send>>,
152        ignore_default_theme: bool,
153    ) -> Window
154    where
155        P: HasWindowHandle,
156        F: Fn(&mut Context),
157        F: 'static + Send,
158    {
159        let window_settings = WindowSettings::new()
160            .with_title(win_desc.title.clone())
161            .with_size(baseview::dpi::LogicalSize::new(
162                // We have our own uniform non-DPI scaling factor that gets applied in addition to
163                // the DPI scaling since both can change independently at runtime
164                win_desc.inner_size.width as f64 * win_desc.user_scale_factor,
165                win_desc.inner_size.height as f64 * win_desc.user_scale_factor,
166            ))
167            .with_parent(Some(parent))
168            .with_fallback_scale_factor(match scale_policy {
169                WindowScalePolicy::ScaleFactor(scale) => Some(scale),
170                WindowScalePolicy::SystemScaleFactor => None,
171            })
172            .with_gl_config(GlConfig::default());
173
174        let window = Window::create(
175            window_settings,
176            move |window: WindowContext| -> Result<ViziaWindow, HandlerError> {
177                Runtime::init_on_ui_thread();
178                let mut cx = Context::new();
179
180                cx.ignore_default_theme = ignore_default_theme;
181                cx.add_built_in_styles();
182
183                let cx = BackendContext::new(cx);
184
185                Ok(ViziaWindow::new(
186                    cx,
187                    win_desc,
188                    frame_policy,
189                    window,
190                    Some(Box::new(app)),
191                    on_idle,
192                    true, // is_parented
193                ))
194            },
195        )
196        .expect("Failed to create baseview window");
197
198        window.show().expect("Failed to show baseview window");
199        window
200    }

202    /// Open a new window that blocks the current thread until the window is destroyed.
203    ///
204    /// * `app` - The Vizia application builder.
205    pub fn open_blocking<F>(
206        win_desc: WindowDescription,
207        scale_policy: WindowScalePolicy,
208        frame_policy: FramePolicy,
209        app: F,
210        on_idle: Option<Box<dyn Fn(&mut Context) + Send>>,
211        ignore_default_theme: bool,
212    ) where
213        F: Fn(&mut Context),
214        F: 'static + Send,
215    {
216        let window_settings = WindowSettings::new()
217            .with_title(win_desc.title.clone())
218            .with_size(baseview::dpi::LogicalSize::new(
219                win_desc.inner_size.width as f64 * win_desc.user_scale_factor,
220                win_desc.inner_size.height as f64 * win_desc.user_scale_factor,
221            ))
222            .with_fallback_scale_factor(match scale_policy {
223                WindowScalePolicy::ScaleFactor(scale) => Some(scale),
224                WindowScalePolicy::SystemScaleFactor => None,
225            })
226            .with_gl_config(GlConfig::default());
227
228        let window = Window::create(
229            window_settings,
230            move |window: WindowContext| -> Result<ViziaWindow, HandlerError> {
231                Runtime::init_on_ui_thread();
232                let mut cx = Context::new();
233
234                cx.ignore_default_theme = ignore_default_theme;
235                cx.add_built_in_styles();
236
237                let cx = BackendContext::new(cx);
238
239                Ok(ViziaWindow::new(
240                    cx,
241                    win_desc,
242                    frame_policy,
243                    window,
244                    Some(Box::new(app)),
245                    on_idle,
246                    false, // is_parented
247                ))
248            },
249        )
250        .expect("Failed to create baseview window");
251
252        window.run_until_closed().expect("baseview event loop failed");
253    }

256impl WindowHandler for ViziaWindow {
257    fn damage(&self, _area: baseview::DamageArea) {
258        // Some hosts dispatch baseview callbacks from a different UI thread than
259        // window construction; ensure this callback thread is registered.
260        Runtime::init_on_ui_thread();
261        if let Ok(mut application) = self.application.try_borrow_mut() {
262            application.mark_damaged();
263        }
264    }
265
266    fn draw(&self) -> Result<(), HandlerError> {
267        // Some hosts dispatch baseview callbacks from a different UI thread than
268        // window construction; ensure this callback thread is registered.
269        Runtime::init_on_ui_thread();
270        let Ok(mut application) = self.application.try_borrow_mut() else {
271            return Ok(());
272        };
273
274        // Run the embedder's idle callback on every frame tick, not only after
275        // an AppKit event. Mirrors what `vizia_winit` does in its
276        // `about_to_wait` handler: the idle callback is the documented hook
277        // for embedders to drain work queued from off-UI threads (e.g. a
278        // plugin host delivering keys on the host thread), and invoking it
279        // only from `on_event` means such work can stall indefinitely when
280        // AppKit events are sparse — observed in a macOS VST3 host where
281        // keystrokes queued from `IPlugView::onKeyDown` only drained when
282        // the next mouse move or key event happened to arrive.
283        application.handle_idle(&self.on_idle);
284
285        // Compute needs redraw after the idle callback so
286        // event queued during on idle now trigger the cb.
287        let needs_redraw = application.on_frame_update();
288
289        application.render();
290
291        // Schedule the next frame according to the configured frame policy.
292        application.schedule_next_frame(needs_redraw);
293
294        Ok(())
295    }
296
297    fn resized(&self, new_size: baseview::WindowSize) -> Result<(), HandlerError> {
298        Runtime::init_on_ui_thread();
299        if let Ok(mut application) = self.application.try_borrow_mut() {
300            application.handle_resized(new_size);
301            application.request_redraw();
302        }
303
304        Ok(())
305    }
306
307    fn on_event(&self, event: Event) -> EventStatus {
308        // Some hosts dispatch baseview callbacks from a different UI thread than
309        // window construction; ensure this callback thread is registered.
310        Runtime::init_on_ui_thread();
311        let Ok(mut application) = self.application.try_borrow_mut() else {
312            return EventStatus::Ignored;
313        };
314
315        let status = application.handle_event(event);
316
317        application.handle_idle(&self.on_idle);
318
319        // Wake the loop so the event is rendered on the next frame.
320        application.request_redraw();
321
322        status
323    }
324}

340pub fn create_surface(
341    size: (i32, i32),
342    fb_info: FramebufferInfo,
343    gr_context: &mut skia_safe::gpu::DirectContext,
344) -> Surface {
345    let backend_render_target = backend_render_targets::make_gl(size, None, 8, fb_info);
346
347    let surface_props = SurfaceProps::new_with_text_properties(
348        SurfacePropsFlags::default(),
349        PixelGeometry::default(),
350        0.5,
351        0.0,
352    );
353
354    gpu::surfaces::wrap_backend_render_target(
355        gr_context,
356        &backend_render_target,
357        SurfaceOrigin::BottomLeft,
358        ColorType::RGBA8888,
359        ColorSpace::new_srgb(),
360        Some(surface_props).as_ref(),
361        // None,
362    )
363    .expect("Could not create skia surface")
