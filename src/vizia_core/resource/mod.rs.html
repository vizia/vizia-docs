48fn fluent_number<'a>(positional: &[FluentValue<'a>], named: &FluentArgs) -> FluentValue<'a> {
49    let Some(first) = positional.first() else {
50        return FluentValue::Error;
51    };
52
53    let mut number = match first {
54        FluentValue::Number(num) => num.clone(),
55        FluentValue::String(value) => value
56            .parse::<FluentNumber>()
57            .unwrap_or_else(|_| FluentNumber::new(0.0, FluentNumberOptions::default())),
58        _ => return FluentValue::Error,
59    };
60
61    number.options.merge(named);
62    FluentValue::Number(number)
63}

72fn datetime_format_pattern(args: &FluentArgs) -> String {
73    let weekday = match style_str(args, "weekday").as_deref() {
74        Some("long") => Some("%A"),
75        Some("short") => Some("%a"),
76        _ => None,
77    };
78
79    let month = match style_str(args, "month").as_deref() {
80        Some("long") => Some("%B"),
81        Some("short") => Some("%b"),
82        Some("2-digit") => Some("%m"),
83        Some("numeric") => Some("%-m"),
84        _ => None,
85    };
86
87    let day = match style_str(args, "day").as_deref() {
88        Some("2-digit") => Some("%d"),
89        Some("numeric") => Some("%-d"),
90        _ => None,
91    };
92
93    let year = match style_str(args, "year").as_deref() {
94        Some("2-digit") => Some("%y"),
95        Some("numeric") => Some("%Y"),
96        _ => None,
97    };
98
99    let hour = match style_str(args, "hour").as_deref() {
100        Some("2-digit") => Some("%H"),
101        Some("numeric") => Some("%-H"),
102        _ => None,
103    };
104
105    let minute = match style_str(args, "minute").as_deref() {
106        Some("2-digit") => Some("%M"),
107        Some("numeric") => Some("%-M"),
108        _ => None,
109    };
110
111    let mut date_parts = Vec::new();
112    if let Some(part) = weekday {
113        date_parts.push(part);
114    }
115    if let Some(part) = month {
116        date_parts.push(part);
117    }
118    if let Some(part) = day {
119        date_parts.push(part);
120    }
121    if let Some(part) = year {
122        date_parts.push(part);
123    }
124
125    let mut pattern = date_parts.join(" ");
126    if hour.is_some() || minute.is_some() {
127        if !pattern.is_empty() {
128            pattern.push(' ');
129        }
130        let mut time_parts = Vec::new();
131        if let Some(part) = hour {
132            time_parts.push(part);
133        }
134        if let Some(part) = minute {
135            time_parts.push(part);
136        }
137        pattern.push_str(&time_parts.join(":"));
138    }
139
140    if pattern.is_empty() { "%Y-%m-%d %H:%M:%S".to_string() } else { pattern }
141}

143fn fluent_datetime<'a>(positional: &[FluentValue<'a>], named: &FluentArgs) -> FluentValue<'a> {
144    let Some(first) = positional.first() else {
145        return FluentValue::Error;
146    };
147
148    let millis = match first {
149        FluentValue::Number(num) => num.value as i64,
150        FluentValue::String(value) => value.parse::<i64>().unwrap_or_default(),
151        _ => return FluentValue::Error,
152    };
153
154    let Ok(timestamp) = Timestamp::from_millisecond(millis) else {
155        return FluentValue::Error;
156    };
157
158    let pattern = datetime_format_pattern(named);
159    match jiff::fmt::strtime::format(&pattern, &timestamp.to_zoned_datetime(TimeZone::UTC)) {
160        Ok(text) => FluentValue::String(text.into()),
161        Err(_) => FluentValue::Error,
162    }
163}

165fn make_bundle(lang: LanguageIdentifier) -> FluentBundle<FluentResource> {
166    let mut bundle = FluentBundle::new(vec![lang]);

168    bundle.add_function("NUMBER", fluent_number).expect("Failed to register NUMBER function");

174/// Structured diagnostics emitted by localization while resolving messages.
175#[derive(Debug, Clone, PartialEq, Eq)]
176pub(crate) enum LocalizationIssue {
177    /// A message key was not found in any fallback bundle.
178    MissingMessage { key: String, requested_locale: String },
179    /// A message attribute was not found in any fallback bundle.
180    MissingAttribute { key: String, attribute: String, requested_locale: String },
181    /// Fluent formatting reported errors while resolving a message.
182    FormatError { key: String, locale: String, details: String },
183}
184
185impl fmt::Display for LocalizationIssue {
186    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
187        match self {
188            LocalizationIssue::MissingMessage { key, requested_locale } => {
189                write!(f, "Missing localized message '{}' for locale '{}'.", key, requested_locale)
190            }
191            LocalizationIssue::MissingAttribute { key, attribute, requested_locale } => write!(
192                f,
193                "Missing localized attribute '{}.{}' for locale '{}'.",
194                key, attribute, requested_locale
195            ),
196            LocalizationIssue::FormatError { key, locale, details } => {
197                write!(f, "Formatting error for key '{}' in locale '{}': {}", key, locale, details)
198            }
199        }
200    }
201}

311/// Trait for loading resources asynchronously or from various sources.
312///
313/// Loaders are invoked in chain-of-responsibility order when a resource is requested.
314/// Return `true` to indicate the request was handled (no further loaders are tried),
315/// or `false` to continue to the next loader.
316pub trait ResourceLoader: Send + Sync + 'static {
317    /// Attempt to load a resource.
318    ///
319    /// `request` describes what to load, while `options` describes how to load it.
320    /// Return `true` if this loader handled the request, `false` to try the next loader.
321    fn load(
322        &self,
323        request: ResourceRequest,
324        options: ResourceLoadOptions,
325        cx: &mut ResourceContext,
326    ) -> bool;
327
328    /// Query the loading status of a resource path.
329    ///
330    /// Default implementation returns `NotLoaded` — override to track async loading progress.
331    fn status(&self, _path: &str) -> LoadingStatus {
332        LoadingStatus::NotLoaded
333    }
334}

436    pub fn renegotiate_language(&mut self) {
437        let available_native = self
438            .translations
439            .keys()
440            .filter(|&x| x != &LanguageIdentifier::default())
441            .collect::<Vec<_>>();
442        let locale_native = sys_locale::get_locale()
443            .and_then(|l| l.parse().ok())
444            .unwrap_or_else(|| available_native.first().copied().cloned().unwrap_or_default());
445        let available = available_native
446            .iter()
447            .map(|locale| locale.to_string().parse::<fluent_langneg::LanguageIdentifier>())
448            .collect::<Result<Vec<_>, _>>()
449            .expect("unic-langid identifiers must parse as ICU locale identifiers");
450        let locale = locale_native
451            .to_string()
452            .parse::<fluent_langneg::LanguageIdentifier>()
453            .expect("unic-langid identifiers must parse as ICU locale identifiers");
454        let default = LanguageIdentifier::default()
455            .to_string()
456            .parse::<fluent_langneg::LanguageIdentifier>()
457            .expect("default locale must parse as an ICU locale identifier");
458        let langs = fluent_langneg::negotiate::negotiate_languages(
459            &[locale],
460            &available,
461            Some(&default),
462            fluent_langneg::NegotiationStrategy::Filtering,
463        );
464        self.language = langs
465            .first()
466            .expect("language negotiation must return its default locale")
467            .to_string()
468            .parse()
469            .expect("negotiated ICU locale must parse as a unic-langid identifier");
470    }

472    fn negotiate_translation_locale(&self, locale: &LanguageIdentifier) -> LanguageIdentifier {
473        if self.translations.contains_key(locale) {
474            return locale.clone();
475        }
476
477        let available = self
478            .translations
479            .keys()
480            .filter(|&lang| lang != &LanguageIdentifier::default())
481            .collect::<Vec<_>>();
482
483        if available.is_empty() {
484            return LanguageIdentifier::default();
485        }
486
487        // Pick a fallback from the registered translations: prefer `self.language` if it
488        // is one of them, otherwise the first registered translation. `available` is
489        // non-empty here (checked above), so `available.first()` is always `Some`.
490        let first_available = *available.first().expect("non-empty checked above");
491        let fallback_native =
492            if available.contains(&&self.language) { &self.language } else { first_available };
493        let available_icu = available
494            .iter()
495            .map(|locale| locale.to_string().parse::<fluent_langneg::LanguageIdentifier>())
496            .collect::<Result<Vec<_>, _>>()
497            .expect("unic-langid identifiers must parse as ICU locale identifiers");
498        let locale_icu = locale
499            .to_string()
500            .parse::<fluent_langneg::LanguageIdentifier>()
501            .expect("unic-langid identifier must parse as an ICU locale identifier");
502        let fallback_icu = fallback_native
503            .to_string()
504            .parse::<fluent_langneg::LanguageIdentifier>()
505            .expect("unic-langid identifier must parse as an ICU locale identifier");
506        let langs = fluent_langneg::negotiate::negotiate_languages(
507            &[locale_icu],
508            &available_icu,
509            Some(&fallback_icu),
510            fluent_langneg::NegotiationStrategy::Filtering,
511        );
512
513        langs
514            .first()
515            .map(|lang| {
516                lang.to_string()
517                    .parse()
518                    .expect("negotiated ICU locale must parse as a unic-langid identifier")
519            })
520            .unwrap_or_else(|| fallback_native.clone())
521    }

523    pub fn translation_locales(&self, locale: &LanguageIdentifier) -> Vec<LanguageIdentifier> {
524        let mut locales = Vec::new();
525
526        if self.translations.contains_key(locale) {
527            locales.push(locale.clone());
528        }
529
530        let negotiated = self.negotiate_translation_locale(locale);
531        if !locales.contains(&negotiated) {
532            locales.push(negotiated);
533        }
534
535        let default = LanguageIdentifier::default();
536        if !locales.contains(&default) {
537            locales.push(default);
538        }
539
540        locales
541    }

543    pub fn add_translation(
544        &mut self,
545        lang: LanguageIdentifier,
546        ftl: String,
547    ) -> Result<(), TranslationError> {
548        match fluent_bundle::FluentResource::try_new(ftl) {
549            Ok(res) => {
550                let bundle =
551                    self.translations.entry(lang.clone()).or_insert_with(|| make_bundle(lang));
552                bundle.add_resource(res).map_err(|errors| {
553                    let msg = format!("{:?}", errors);
554                    TranslationError::BundleError(msg)
555                })?;
556                self.renegotiate_language();
557                Ok(())
558            }
559            Err((_, parse_errors)) => {
560                let msg =
561                    parse_errors.iter().map(|e| format!("{:?}", e)).collect::<Vec<_>>().join("; ");
562                Err(TranslationError::InvalidFtl(msg))
563            }
564        }
565    }

581    pub fn evict_unused_images(&mut self) {
582        let rem = self
583            .images
584            .iter()
585            .filter_map(|(id, img)| match img.retention_policy {
586                ImageRetentionPolicy::DropWhenUnusedForOneFrame => (!img.used).then_some(*id),
587
588                ImageRetentionPolicy::DropWhenNoObservers => {
589                    img.observers.is_empty().then_some(*id)
590                }
591
592                ImageRetentionPolicy::Forever => None,
593            })
594            .collect::<Vec<_>>();
595
596        for id in rem {
597            self.images.remove(&id);
598
599            // Collect all name→id mappings that point to this id.
600            let removed_names = self
601                .image_ids
602                .iter()
603                .filter_map(|(name, img_id)| (*img_id == id).then_some(name.clone()))
604                .collect::<Vec<_>>();
605
606            // Remove each name from image_ids and image_sources, recording the resolved
607            // source paths (image_sources maps name→path; for direct-path references the
608            // name IS the path, so fall back to the name itself).
609            let mut source_paths = Vec::new();
610            for name in removed_names {
611                self.image_ids.remove(&name);
612                let source_path = self.image_sources.remove(&name).unwrap_or(name);
613                source_paths.push(source_path);
614            }
615
616            // Clear loading_status for each source path, but only when no remaining
617            // image_sources entry still references it (another name may alias the same path).
618            for source_path in source_paths {
619                if !self.image_sources.values().any(|p| p == &source_path) {
620                    self.loading_status.remove(&source_path);
621                }
622            }
623
624            self.image_id_manager.destroy(id);
625        }
626    }

628    /// Query the loading status of a resource path.
629    pub fn resource_status(&self, path: &str) -> LoadingStatus {
630        // First check cached status signal
631        if let Some(signal) = self.loading_status.get(path) {
632            return signal.get();
633        }
634
635        // If not in cache, ask each loader in the chain
636        for loader in &self.resource_loaders {
637            let status = loader.status(path);
638            if status != LoadingStatus::NotLoaded {
639                return status;
640            }
641        }
642
643        LoadingStatus::NotLoaded
644    }

762    fn format_datetime(
763        millis: i64,
764        options: &[(&'static str, &'static str)],
765    ) -> FluentValue<'static> {
766        let mut args = FluentArgs::new();
767        for (key, value) in options {
768            args.set(*key, *value);
769        }
770        fluent_datetime(&[FluentValue::from(millis)], &args)
771    }

774        FluentValue::String(text.to_owned().into())

896    fn datetime_formats_negative_years() {
897        // 0000-06-15, -0001-06-15 and -9999-06-15 at noon UTC.
898        let year_zero = -62_152_833_600_000;
899        let year_minus_one = -62_184_456_000_000;
900        let year_minus_9999 = -377_690_817_600_000;
901
902        assert_eq!(format_datetime(year_zero, &[("year", "numeric")]), string_value("0000"));
903        assert_eq!(format_datetime(year_zero, &[("year", "2-digit")]), string_value("00"));
904        assert_eq!(format_datetime(year_minus_one, &[("year", "numeric")]), string_value("-001"));
905        assert_eq!(format_datetime(year_minus_one, &[("year", "2-digit")]), string_value("-1"));
906        assert_eq!(format_datetime(year_minus_9999, &[("year", "numeric")]), string_value("-9999"));
907        assert_eq!(format_datetime(year_minus_9999, &[("year", "2-digit")]), string_value("-99"));
908    }
