5macro_rules! process_auto_animations {
6    ($cx:expr, $property:expr, $height:expr) => {
7        if let Some(animations) = $property.get_active_animations() {
8            let mut entities = vec![];
9
10            for animation in animations {
11                if animation.keyframes.iter().any(|keyframe| keyframe.value == Units::Auto) {
12                    for entity in animation.entities.iter() {
13                        let current_bounds = $cx.cache.get_bounds(*entity);
14                        let current_measured =
15                            if $height { current_bounds.h } else { current_bounds.w };
16                        entities.push((*entity, animation.clone(), current_measured));
17                    }
18                }
19            }
20
21            if entities.is_empty() {
22                // No auto keyframes for this property in the current frame.
23            } else {
24                // Resolve auto values against a root layout pass so wrapped text is measured using
25                // real parent constraints (especially width) rather than isolated node layout.
26                for (entity, animation, _) in entities.iter() {
27                    $property.stop_animation(*entity, animation.id);
28                    $property.insert(*entity, Units::Auto);
29                }
30
31                Entity::root().layout(
32                    &mut $cx.cache,
33                    &$cx.tree,
34                    &$cx.style,
35                    &mut SubLayout {
36                        text_context: &mut $cx.text_context,
37                        resource_manager: &$cx.resource_manager,
38                    },
39                );
40
41                for (entity, mut animation, current_measured) in entities {
42                    $property.remove(entity);
43
44                    let measured_target =
45                        if let Some(bounds) = $cx.cache.relative_bounds.get(entity) {
46                            if $height { bounds.h } else { bounds.w }
47                        } else {
48                            let bounds = $cx.cache.get_bounds(entity);
49                            if $height { bounds.h } else { bounds.w }
50                        };
51
52                    animation.keyframes.iter_mut().for_each(|keyframe| {
53                        if keyframe.value == Units::Auto {
54                            // Preserve transition direction: start keyframes resolve from current
55                            // geometry, later keyframes resolve from target auto geometry.
56                            let measured = if keyframe.time <= 0.0 {
57                                current_measured
58                            } else {
59                                measured_target
60                            };
61                            keyframe.value = Units::Pixels(measured);
62                        }
63                    });
64
65                    let id = $cx.style.animation_manager.create();
66                    $property.insert_animation(id, animation.clone());
67                    $property.play_animation(
68                        entity,
69                        id,
70                        animation.start_time,
71                        animation.duration,
72                        animation.delay,
73                    );
74                }
75            }
76        }
77    };
78}

80pub(crate) fn animation_system(cx: &mut Context) -> bool {
81    cx.style.play_pending_animations();
82
83    process_auto_animations!(cx, cx.style.max_height, true);
84    process_auto_animations!(cx, cx.style.max_width, false);
85    process_auto_animations!(cx, cx.style.height, true);
86    process_auto_animations!(cx, cx.style.width, false);
87
88    // Tick all animations
89
90    let time = Instant::now();
91
92    let mut redraw_entities = Vec::new();
93    let mut reflow_entities = Vec::new();
94    let mut relayout_entities = Vec::new();
95    let mut retransform_entities = Vec::new();
96    let mut reclip_entities = Vec::new();
97
98    // Properties which affect rendering
99    // Opacity
100    redraw_entities.extend(cx.style.opacity.tick(time));
101    // Corner Colour
102    redraw_entities.extend(cx.style.border_top_color.tick(time));
103    redraw_entities.extend(cx.style.border_right_color.tick(time));
104    redraw_entities.extend(cx.style.border_bottom_color.tick(time));
105    redraw_entities.extend(cx.style.border_left_color.tick(time));
106    // Corner Radius
107    redraw_entities.extend(cx.style.corner_top_left_radius.tick(time));
108    redraw_entities.extend(cx.style.corner_top_right_radius.tick(time));
109    redraw_entities.extend(cx.style.corner_bottom_left_radius.tick(time));
110    redraw_entities.extend(cx.style.corner_bottom_right_radius.tick(time));
111    // Background
112    redraw_entities.extend(cx.style.background_color.tick(time));
113    redraw_entities.extend(cx.style.background_image.tick(time));
114    redraw_entities.extend(cx.style.background_size.tick(time));
115    // Box Shadow
116    redraw_entities.extend(cx.style.shadow.tick(time));
117    // Transform
118    retransform_entities.extend(cx.style.transform.tick(time));
119    retransform_entities.extend(cx.style.transform_origin.tick(time));
120    retransform_entities.extend(cx.style.translate.tick(time));
121    retransform_entities.extend(cx.style.rotate.tick(time));
122    retransform_entities.extend(cx.style.scale.tick(time));
123    // Outline
124    redraw_entities.extend(cx.style.outline_color.tick(time));
125    redraw_entities.extend(cx.style.outline_offset.tick(time));
126    redraw_entities.extend(cx.style.outline_width.tick(time));
127    // Clip Path
128    reclip_entities.extend(cx.style.clip_path.tick(time));
129
130    redraw_entities.extend(cx.style.fill.tick(time));
131
132    // Font Color
133    reflow_entities.extend(cx.style.font_color.tick(time));
134    // Font Size
135    reflow_entities.extend(cx.style.font_size.tick(time));
136    // Letter Spacing
137    reflow_entities.extend(cx.style.letter_spacing.tick(time));
138    // Line Height
139    reflow_entities.extend(cx.style.line_height.tick(time));
140
141    // Properties which affect layout
142    relayout_entities.extend(cx.style.display.tick(time));
143    // Border Width
144    relayout_entities.extend(cx.style.border_top_width.tick(time));
145    relayout_entities.extend(cx.style.border_right_width.tick(time));
146    relayout_entities.extend(cx.style.border_bottom_width.tick(time));
147    relayout_entities.extend(cx.style.border_left_width.tick(time));
148    // Space
149    relayout_entities.extend(cx.style.left.tick(time));
150    relayout_entities.extend(cx.style.right.tick(time));
151    relayout_entities.extend(cx.style.top.tick(time));
152    relayout_entities.extend(cx.style.bottom.tick(time));
153    // Size
154    relayout_entities.extend(cx.style.width.tick(time));
155    relayout_entities.extend(cx.style.height.tick(time));
156    // Min/Max Size
157    relayout_entities.extend(cx.style.max_width.tick(time));
158    relayout_entities.extend(cx.style.max_height.tick(time));
159    relayout_entities.extend(cx.style.min_width.tick(time));
160    relayout_entities.extend(cx.style.min_height.tick(time));
161    // Min/Max Gap
162    relayout_entities.extend(cx.style.max_horizontal_gap.tick(time));
163    relayout_entities.extend(cx.style.max_vertical_gap.tick(time));
164    relayout_entities.extend(cx.style.min_horizontal_gap.tick(time));
165    relayout_entities.extend(cx.style.min_vertical_gap.tick(time));
166    // Row/Col Between
167    relayout_entities.extend(cx.style.vertical_gap.tick(time));
168    relayout_entities.extend(cx.style.horizontal_gap.tick(time));
169    // Child Space
170    relayout_entities.extend(cx.style.padding_left.tick(time));
171    relayout_entities.extend(cx.style.padding_right.tick(time));
172    relayout_entities.extend(cx.style.padding_top.tick(time));
173    relayout_entities.extend(cx.style.padding_bottom.tick(time));
174
175    // Tick animations on custom color properties
176    for store in cx.style.custom_color_props.values_mut() {
177        redraw_entities.extend(store.tick(time));
178    }
179    // Tick animations on custom length properties
180    for store in cx.style.custom_length_props.values_mut() {
181        redraw_entities.extend(store.tick(time));
182    }
183    // Tick animations on custom font-size properties
184    for store in cx.style.custom_font_size_props.values_mut() {
185        reflow_entities.extend(store.tick(time));
186    }
187    // Tick animations on custom letter-spacing properties
188    for store in cx.style.custom_letter_spacing_props.values_mut() {
189        reflow_entities.extend(store.tick(time));
190    }
191    // Tick animations on custom line-height properties
192    for store in cx.style.custom_line_height_props.values_mut() {
193        reflow_entities.extend(store.tick(time));
194    }
195    // Tick animations on custom units properties
196    for store in cx.style.custom_units_props.values_mut() {
197        relayout_entities.extend(store.tick(time));
198    }
199    // Tick animations on custom opacity properties
200    for store in cx.style.custom_opacity_props.values_mut() {
201        redraw_entities.extend(store.tick(time));
202    }
203
204    for entity in relayout_entities.iter() {
205        cx.style.needs_relayout(*entity);
206    }
207
208    for entity in redraw_entities.iter() {
209        cx.needs_redraw(*entity);
210    }
211
212    for entity in reflow_entities.iter() {
213        cx.style.text_construction.insert(*entity);
214    }
215
216    for entity in retransform_entities.iter() {
217        cx.needs_retransform(*entity);
218        cx.needs_redraw(*entity);
219    }
220
221    for entity in reclip_entities.iter() {
222        cx.needs_reclip(*entity);
223        cx.needs_redraw(*entity);
224    }
225
226    !redraw_entities.is_empty()
227        | !relayout_entities.is_empty()
228        | !reflow_entities.is_empty()
229        | !retransform_entities.is_empty()
230        | !reclip_entities.is_empty()
231}
