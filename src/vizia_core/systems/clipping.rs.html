11/// Computes the clipping path for each entity in the layout tree.
12pub(crate) fn clipping_system(cx: &mut Context) {
13    if cx.style.reclip.is_empty() {
14        return;
15    }
16
17    let dirty_subtree_roots = minimal_layout_dirty_roots(&cx.tree, &cx.style.reclip);
18    let mut invalidated_subtrees = HashSet::new();
19
20    for &root in &dirty_subtree_roots {
21        let iter = LayoutTreeIterator::subtree(&cx.tree, root);
22        for entity in iter {
23            if !cx.style.reclip.contains(&entity) {
24                continue;
25            }
26
27            let parent = cx.tree.get_layout_parent(entity).unwrap_or(Entity::root());
28
29            let Some(bounds) = cx.cache.bounds.get(entity).copied() else { continue };
30            let previous_clip_bounds = cx
31                .cache
32                .clip_path
33                .get(entity)
34                .and_then(|clip_path| clip_path.as_ref())
35                .map(|clip_path| *clip_path.bounds());
36
37            if entity == Entity::root() {
38                let clip_path = build_clip_path(cx, Entity::root(), bounds);
39                cx.cache.clip_path.insert(entity, Some(clip_path));
40                let new_clip_bounds = cx
41                    .cache
42                    .clip_path
43                    .get(entity)
44                    .and_then(|clip_path| clip_path.as_ref())
45                    .map(|clip_path| *clip_path.bounds());
46
47                let should_invalidate_subtree = should_invalidate_subtree(
48                    cx,
49                    entity,
50                    &dirty_subtree_roots,
51                    &invalidated_subtrees,
52                );
53
54                if previous_clip_bounds != new_clip_bounds && should_invalidate_subtree {
55                    for descendant in LayoutTreeIterator::subtree(&cx.tree, entity).skip(1) {
56                        cx.cache.draw_bounds.remove(descendant);
57                    }
58                    invalidated_subtrees.insert(entity);
59                }
60                continue;
61            }
62
63            let overflowx = cx.style.overflowx.get(entity).copied().unwrap_or_default();
64            let overflowy = cx.style.overflowy.get(entity).copied().unwrap_or_default();
65
66            let scale = cx.style.scale_factor();
67
68            let transform = cx
69                .cache
70                .transform
71                .get(entity)
72                .copied()
73                .unwrap_or(skia_safe::Matrix::new_identity());
74
75            let shape_clip_path = cx.style.clip_path.get(entity).and_then(|clip| match clip {
76                ClipPath::Auto => None,
77                ClipPath::Shape(rect) => {
78                    let clip_bounds = bounds.shrink_sides(
79                        rect.3.to_pixels(bounds.w, scale),
80                        rect.0.to_pixels(bounds.h, scale),
81                        rect.1.to_pixels(bounds.w, scale),
82                        rect.2.to_pixels(bounds.h, scale),
83                    );
84
85                    Some(build_clip_path(cx, entity, clip_bounds))
86                }
87            });
88
89            let window_entity = if cx.tree.is_window(entity) {
90                entity
91            } else {
92                cx.tree.get_parent_window(entity).unwrap_or(Entity::root())
93            };
94            let window_bounds = cx.cache.bounds.get(window_entity).copied().unwrap_or(bounds);
95
96            let overflow_clip_path = match (overflowx, overflowy) {
97                (Overflow::Visible, Overflow::Visible) => None,
98                (Overflow::Hidden, Overflow::Visible) => {
99                    let left = bounds.left();
100                    let right = bounds.right();
101                    let top = window_bounds.top();
102                    let bottom = window_bounds.bottom();
103                    let clip_bounds = BoundingBox::from_min_max(left, top, right, bottom);
104                    Some(build_clip_path(cx, entity, clip_bounds))
105                }
106                (Overflow::Visible, Overflow::Hidden) => {
107                    let left = window_bounds.left();
108                    let right = window_bounds.right();
109                    let top = bounds.top();
110                    let bottom = bounds.bottom();
111                    let clip_bounds = BoundingBox::from_min_max(left, top, right, bottom);
112                    Some(build_clip_path(cx, entity, clip_bounds))
113                }
114                (Overflow::Hidden, Overflow::Hidden) => Some(build_clip_path(cx, entity, bounds)),
115            };
116
117            let clip_path = match (overflow_clip_path, shape_clip_path) {
118                (Some(overflow_path), Some(shape_path)) => {
119                    overflow_path.op(&shape_path, skia_safe::PathOp::Intersect)
120                }
121                (Some(overflow_path), None) => Some(overflow_path),
122                (None, Some(shape_path)) => Some(shape_path),
123                (None, None) => None,
124            };
125
126            let clip_path = clip_path.map(|clip_path| clip_path.make_transform(&transform));
127
128            let ignore_clipping = cx.style.ignore_clipping.get(entity).copied().unwrap_or(false);
129
130            let parent_clip_path = if ignore_clipping || cx.tree.is_window(entity) {
131                None
132            } else {
133                cx.cache.clip_path.get(parent).cloned().flatten()
134            };
135
136            let effective_clip_path = match (clip_path, parent_clip_path) {
137                (Some(clip_path), Some(parent_clip_path)) => {
138                    clip_path.op(&parent_clip_path, skia_safe::PathOp::Intersect)
139                }
140                (Some(clip_path), None) => Some(clip_path),
141                (None, Some(parent_clip_path)) => Some(parent_clip_path),
142                (None, None) => None,
143            };
144
145            let new_clip_bounds = effective_clip_path.as_ref().map(|clip_path| *clip_path.bounds());
146            cx.cache.clip_path.insert(entity, effective_clip_path);
147            let should_invalidate_subtree =
148                should_invalidate_subtree(cx, entity, &dirty_subtree_roots, &invalidated_subtrees);
149
150            if previous_clip_bounds != new_clip_bounds && should_invalidate_subtree {
151                for descendant in LayoutTreeIterator::subtree(&cx.tree, entity).skip(1) {
152                    cx.cache.draw_bounds.remove(descendant);
153                }
154                invalidated_subtrees.insert(entity);
155            }
156        }
157    }
158
159    cx.style.reclip.clear();
160}

162fn should_invalidate_subtree(
163    cx: &Context,
164    entity: Entity,
165    dirty_subtree_roots: &HashSet<Entity>,
166    invalidated_subtrees: &HashSet<Entity>,
167) -> bool {
168    dirty_subtree_roots.contains(&entity)
169        || !has_invalidated_layout_ancestor(cx, entity, invalidated_subtrees)
170}
171
172fn has_invalidated_layout_ancestor(
173    cx: &Context,
174    entity: Entity,
175    invalidated_subtrees: &HashSet<Entity>,
176) -> bool {
177    let mut ancestor = cx.tree.get_layout_parent(entity);
178
179    while let Some(current) = ancestor {
180        if invalidated_subtrees.contains(&current) {
181            return true;
182        }
183
184        ancestor = cx.tree.get_layout_parent(current);
185    }
186
187    false
188}

190fn build_clip_path(cx: &Context, entity: Entity, clip_bounds: BoundingBox) -> skia_safe::Path {
191    let outset = (0.0, 0.0);
192    let bounds = cx.cache.bounds.get(entity).copied().unwrap_or_default();
193    let scale = cx.style.scale_factor();
194    let corner_top_left_radius = cx
195        .style
196        .corner_top_left_radius
197        .get(entity)
198        .map(|l| l.to_pixels(bounds.w.min(bounds.h), scale))
199        .unwrap_or_default();
200    let corner_top_right_radius = cx
201        .style
202        .corner_top_right_radius
203        .get(entity)
204        .map(|l| l.to_pixels(bounds.w.min(bounds.h), scale))
205        .unwrap_or_default();
206    let corner_bottom_right_radius = cx
207        .style
208        .corner_bottom_right_radius
209        .get(entity)
210        .map(|l| l.to_pixels(bounds.w.min(bounds.h), scale))
211        .unwrap_or_default();
212    let corner_bottom_left_radius = cx
213        .style
214        .corner_bottom_left_radius
215        .get(entity)
216        .map(|l| l.to_pixels(bounds.w.min(bounds.h), scale))
217        .unwrap_or_default();
218
219    let corner_top_left_shape =
220        cx.style.corner_top_left_shape.get(entity).copied().unwrap_or_default();
221    let corner_top_right_shape =
222        cx.style.corner_top_right_shape.get(entity).copied().unwrap_or_default();
223    let corner_bottom_right_shape =
224        cx.style.corner_bottom_right_shape.get(entity).copied().unwrap_or_default();
225    let corner_bottom_left_shape =
226        cx.style.corner_bottom_left_shape.get(entity).copied().unwrap_or_default();
227
228    let corner_top_left_smoothing =
229        cx.style.corner_top_left_smoothing.get(entity).copied().unwrap_or(0.0);
230    let corner_top_right_smoothing =
231        cx.style.corner_top_right_smoothing.get(entity).copied().unwrap_or(0.0);
232    let corner_bottom_right_smoothing =
233        cx.style.corner_bottom_right_smoothing.get(entity).copied().unwrap_or(0.0);
234    let corner_bottom_left_smoothing =
235        cx.style.corner_bottom_left_smoothing.get(entity).copied().unwrap_or(0.0);
236
237    let bounds = clip_bounds;
238
239    let bounds = BoundingBox::from_min_max(0.0, 0.0, bounds.w, bounds.h);
240
241    let rect: Rect = bounds.into();
242
243    let mut rr = RRect::new_rect_radii(
244        rect,
245        &[
246            Point::new(corner_top_left_radius, corner_top_left_radius),
247            Point::new(corner_top_right_radius, corner_top_right_radius),
248            Point::new(corner_bottom_right_radius, corner_bottom_right_radius),
249            Point::new(corner_bottom_left_radius, corner_bottom_left_radius),
250        ],
251    );
252
253    rr = rr.with_outset(outset);
254
255    let x = rr.bounds().x();
256    let y = rr.bounds().y();
257    let width = rr.width();
258    let height = rr.height();
259
260    //TODO: Cache the path and regenerate if the bounds change
261    let mut path = PathBuilder::new();
262
263    if width == height
264        && corner_bottom_left_radius == width / 2.0
265        && corner_bottom_right_radius == width / 2.0
266        && corner_top_left_radius == height / 2.0
267        && corner_top_right_radius == height / 2.0
268    {
269        path.add_circle((width / 2.0, bounds.h / 2.0), width / 2.0, PathDirection::CW);
270    } else if corner_top_left_radius == corner_top_right_radius
271        && corner_top_right_radius == corner_bottom_right_radius
272        && corner_bottom_right_radius == corner_bottom_left_radius
273        && corner_top_left_smoothing == 0.0
274        && corner_top_left_smoothing == corner_top_right_smoothing
275        && corner_top_right_smoothing == corner_bottom_right_smoothing
276        && corner_bottom_right_smoothing == corner_bottom_left_smoothing
277        && corner_top_left_shape == CornerShape::Round
278        && corner_top_left_shape == corner_top_right_shape
279        && corner_top_right_shape == corner_bottom_right_shape
280        && corner_bottom_right_shape == corner_bottom_left_shape
281    {
282        path.add_rrect(rr, None, None);
283    } else {
284        let top_right = rr.radii(Corner::UpperRight).x;
285
286        if top_right > 0.0 {
287            let (a, b, c, d, l, p, radius) = compute_smooth_corner(
288                top_right,
289                corner_top_right_smoothing,
290                bounds.width(),
291                bounds.height(),
292            );
293
294            path.move_to((f32::max(width / 2.0, width - p), 0.0));
295            if corner_top_right_shape == CornerShape::Round {
296                path.cubic_to(
297                    (width - (p - a), 0.0),
298                    (width - (p - a - b), 0.0),
299                    (width - (p - a - b - c), d),
300                )
301                .r_arc_to((radius, radius), 0.0, ArcSize::Small, PathDirection::CW, (l, l))
302                .cubic_to(
303                    (width, p - a - b),
304                    (width, p - a),
305                    (width, f32::min(height / 2.0, p)),
306                );
307            } else {
308                path.line_to((width, f32::min(height / 2.0, p)));
309            }
310        } else {
311            path.move_to((width / 2.0, 0.0)).line_to((width, 0.0)).line_to((width, height / 2.0));
312        }
313
314        let bottom_right = rr.radii(Corner::LowerRight).x;
315        if bottom_right > 0.0 {
316            let (a, b, c, d, l, p, radius) =
317                compute_smooth_corner(bottom_right, corner_bottom_right_smoothing, width, height);
318
319            path.line_to((width, f32::max(height / 2.0, height - p)));
320            if corner_bottom_right_shape == CornerShape::Round {
321                path.cubic_to(
322                    (width, height - (p - a)),
323                    (width, height - (p - a - b)),
324                    (width - d, height - (p - a - b - c)),
325                )
326                .r_arc_to((radius, radius), 0.0, ArcSize::Small, PathDirection::CW, (-l, l))
327                .cubic_to(
328                    (width - (p - a - b), height),
329                    (width - (p - a), height),
330                    (f32::max(width / 2.0, width - p), height),
331                );
332            } else {
333                path.line_to((f32::max(width / 2.0, width - p), height));
334            }
335        } else {
336            path.line_to((width, height)).line_to((width / 2.0, height));
337        }
338
339        let bottom_left = rr.radii(Corner::LowerLeft).x;
340        if bottom_left > 0.0 {
341            let (a, b, c, d, l, p, radius) =
342                compute_smooth_corner(bottom_left, corner_bottom_left_smoothing, width, height);
343
344            path.line_to((f32::min(width / 2.0, p), height));
345            if corner_bottom_left_shape == CornerShape::Round {
346                path.cubic_to((p - a, height), (p - a - b, height), (p - a - b - c, height - d))
347                    .r_arc_to((radius, radius), 0.0, ArcSize::Small, PathDirection::CW, (-l, -l))
348                    .cubic_to(
349                        (0.0, height - (p - a - b)),
350                        (0.0, height - (p - a)),
351                        (0.0, f32::max(height / 2.0, height - p)),
352                    );
353            } else {
354                path.line_to((0.0, f32::max(height / 2.0, height - p)));
355            }
356        } else {
357            path.line_to((0.0, height)).line_to((0.0, height / 2.0));
358        }
359
360        let top_left = rr.radii(Corner::UpperLeft).x;
361        if top_left > 0.0 {
362            let (a, b, c, d, l, p, radius) =
363                compute_smooth_corner(top_left, corner_top_left_smoothing, width, height);
364
365            path.line_to((0.0, f32::min(height / 2.0, p)));
366            if corner_top_left_shape == CornerShape::Round {
367                path.cubic_to((0.0, p - a), (0.0, p - a - b), (d, p - a - b - c))
368                    .r_arc_to((radius, radius), 0.0, ArcSize::Small, PathDirection::CW, (l, -l))
369                    .cubic_to((p - a - b, 0.0), (p - a, 0.0), (f32::min(width / 2.0, p), 0.0));
370            } else {
371                path.line_to((f32::min(width / 2.0, p), 0.0));
372            }
373        } else {
374            path.line_to((0.0, 0.0));
375        }
376
377        path.close();
378
379        path.offset((x, y));
380    }
381
382    path.offset(clip_bounds.top_left());
383
384    path.detach()
385}

387// Helper function for computing a rounded corner with variable smoothing
388fn compute_smooth_corner(
389    corner_radius: f32,
390    smoothing: f32,
391    width: f32,
392    height: f32,
393) -> (f32, f32, f32, f32, f32, f32, f32) {
394    let max_p = f32::min(width, height) / 2.0;
395    let corner_radius = f32::min(corner_radius, max_p);
396
397    let p = f32::min((1.0 + smoothing) * corner_radius, max_p);
398
399    let angle_alpha: f32;
400    let angle_beta: f32;
401
402    if corner_radius <= max_p / 2.0 {
403        angle_alpha = 45.0 * smoothing;
404        angle_beta = 90.0 * (1.0 - smoothing);
405    } else {
406        let diff_ratio = (corner_radius - max_p / 2.0) / (max_p / 2.0);
407
408        angle_alpha = 45.0 * smoothing * (1.0 - diff_ratio);
409        angle_beta = 90.0 * (1.0 - smoothing * (1.0 - diff_ratio));
410    }
411
412    let angle_theta = (90.0 - angle_beta) / 2.0;
413    let dist_p3_p4 = corner_radius * (angle_theta / 2.0).to_radians().tan();
414
415    let l = (angle_beta / 2.0).to_radians().sin() * corner_radius * SQRT_2;
416    let c = dist_p3_p4 * angle_alpha.to_radians().cos();
417    let d = c * angle_alpha.to_radians().tan();
418    let b = (p - l - c - d) / 3.0;
419    let a = 2.0 * b;
420
421    (a, b, c, d, l, p, corner_radius)
