9pub(crate) fn draw_system(
10    cx: &mut Context,
11    window_entity: Entity,
12    surface: &mut Surface,
13    dirty_surface: &mut Surface,
14) -> bool {
15    if cx.windows.is_empty() {
16        return false;
17    }
18
19    if !cx.entity_manager.is_alive(window_entity) {
20        return false;
21    }
22
23    let window = cx.windows.get_mut(&window_entity).unwrap();
24
25    let mut dirty_rect = std::mem::take(&mut window.dirty_rect);
26    let redraw_list = std::mem::take(&mut window.redraw_list);
27
28    // Note: dirty_rect can be populated by entity removal independently of
29    // redraw_list, and backdrop filter processing generates dirty areas outside
30    // the redraw_list. Un-commenting would cause rendering artifacts. Maybe we
31    // can revisit this in the future.
32    // if redraw_list.is_empty() {
33    //     return false;
34    // }
35
36    for &entity in &redraw_list {
37        // Skip binding views
38        if cx.tree.is_ignored(entity) {
39            continue;
40        }
41
42        if cx.tree.is_window(entity) && entity != window_entity {
43            continue;
44        }
45
46        if entity.visible(&cx.style) {
47            let draw_bounds = draw_bounds(&cx.style, &cx.cache, &cx.tree, entity);
48
49            let mut dirty_bounds = draw_bounds;
50
51            if let Some(previous_draw_bounds) = cx.cache.draw_bounds.get(entity) {
52                dirty_bounds = dirty_bounds.union(previous_draw_bounds);
53            }
54
55            if dirty_bounds.w != 0.0 && dirty_bounds.h != 0.0 {
56                if let Some(dr) = &mut dirty_rect {
57                    *dr = dr.union(&dirty_bounds);
58                } else {
59                    dirty_rect = Some(dirty_bounds);
60                }
61            }
62
63            if let Some(dr) = cx.cache.draw_bounds.get_mut(entity) {
64                *dr = draw_bounds;
65            } else {
66                cx.cache.draw_bounds.insert(entity, draw_bounds);
67            }
68        }
69    }
70
71    let iter = LayoutTreeIterator::full(&cx.tree);
72    for entity in iter {
73        if cx.tree.is_ignored(entity) {
74            continue;
75        }
76
77        if cx.tree.is_window(entity) && entity != window_entity {
78            continue;
79        }
80
81        // Check if the entity has a backdrop filter style.
82        if cx.style.backdrop_filter.get(entity).is_some() {
83            if entity.visible(&cx.style) {
84                // Entity is VISIBLE and has a backdrop filter.
85                // Skip recomputation if already processed in redraw_list.
86                if !redraw_list.contains(&entity) {
87                    let bf_current_bounds = draw_bounds(&cx.style, &cx.cache, &cx.tree, entity);
88
89                    // Update cache for visible entity
90                    if let Some(dr) = cx.cache.draw_bounds.get_mut(entity) {
91                        *dr = bf_current_bounds;
92                    } else {
93                        cx.cache.draw_bounds.insert(entity, bf_current_bounds);
94                    }
95
96                    if bf_current_bounds.w > 0.0 && bf_current_bounds.h > 0.0 {
97                        // Ensure bounds are valid
98                        // Condition to update dirty_rect:
99                        // 1. dirty_rect is None (then set it to bf_current_bounds).
100                        // 2. dirty_rect is Some, and bf_current_bounds intersects with it (then union).
101                        if dirty_rect.is_none_or(|current_dr_val| {
102                            bf_current_bounds.intersects(&current_dr_val)
103                        }) {
104                            dirty_rect =
105                                Some(dirty_rect.map_or(bf_current_bounds, |current_dr_val| {
106                                    current_dr_val.union(&bf_current_bounds)
107                                }));
108                        }
109                    }
110                }
111            } else {
112                // Entity is INVISIBLE but has (or had) a backdrop filter style.
113                // Its *previous* bounds need to be added to dirty_rect.
114                if let Some(previous_draw_bounds) = cx.cache.draw_bounds.get(entity).copied() {
115                    if previous_draw_bounds.w > 0.0 && previous_draw_bounds.h > 0.0 {
116                        // Ensure previous bounds were valid
117                        // Union previous_draw_bounds into dirty_rect.
118                        // If dirty_rect is None, it becomes Some(previous_draw_bounds).
119                        // If dirty_rect is Some, it's unioned with previous_draw_bounds.
120                        dirty_rect =
121                            Some(dirty_rect.map_or(previous_draw_bounds, |current_dr_val| {
122                                current_dr_val.union(&previous_draw_bounds)
123                            }));
124                    }
125                }
126
127                // Remove from cache as it's no longer visible with these bounds.
128                cx.cache.draw_bounds.remove(entity);
129            }
130        }
131    }
132
133    // if dirty_rect.is_none() {
134    //     return true;
135    // }
136
137    let canvas = dirty_surface.canvas();
138
139    canvas.save();
140
141    if let Some(rect) = dirty_rect.map(Rect::from) {
142        canvas.clip_rect(rect, ClipOp::Intersect, false);
143        canvas.clear(Color::transparent());
144    }
145
146    cx.resource_manager.mark_images_unused();
147
148    let mut queue = BinaryHeap::new();
149    queue.push(ZEntity { index: 0, entity: window_entity, visible: true });
150
151    while let Some(zentity) = queue.pop() {
152        canvas.save();
153        draw_entity(
154            &mut DrawContext {
155                current: zentity.entity,
156                style: &cx.style,
157                cache: &mut cx.cache,
158                tree: &cx.tree,
159                models: &cx.models,
160                views: &mut cx.views,
161                resource_manager: &cx.resource_manager,
162                text_context: &mut cx.text_context,
163                modifiers: &cx.modifiers,
164                mouse: &cx.mouse,
165                windows: &mut cx.windows,
166            },
167            &dirty_rect,
168            canvas,
169            zentity.index,
170            &mut queue,
171            zentity.visible,
172        );
173        canvas.restore();
174    }
175
176    canvas.restore();
177
178    // surface.canvas().clear(Color::transparent());
179    dirty_surface.draw(surface.canvas(), (0, 0), SamplingOptions::default(), None);
180
181    // Debug draw dirty rect
182    // if let Some(rect) = dirty_rect.map(Rect::from) {
183    //     let mut paint = Paint::default();
184    //     paint.set_style(skia_safe::PaintStyle::Stroke);
185    //     paint.set_color(Color::red());
186    //     paint.set_stroke_width(1.0);
187    //     surface.canvas().draw_rect(rect, &paint);
188    // }
189
190    true
191}

193fn draw_entity(
194    cx: &mut DrawContext,
195    dirty_rect: &Option<BoundingBox>,
196    canvas: &Canvas,
197    current_z: i32,
198    queue: &mut BinaryHeap<ZEntity>,
199    visible: bool,
200) {
201    let current = cx.current;
202
203    // Skip views with display: none.
204    if cx.display() == Display::None {
205        return;
206    }
207
208    let z_index = cx.z_index();
209
210    if z_index > current_z {
211        queue.push(ZEntity { index: z_index, entity: current, visible });
212        return;
213    }
214
215    let backdrop_filter = cx.backdrop_filter();
216    let blend_mode = cx.style.blend_mode.get(current).copied().unwrap_or_default();
217
218    let layer_count =
219        if cx.opacity() != 1.0 || backdrop_filter.is_some() || blend_mode != BlendMode::Normal {
220            let mut paint = Paint::default();
221            paint.set_alpha_f(cx.opacity());
222            paint.set_blend_mode(blend_mode.into());
223
224            let rect: Rect = cx.bounds().into();
225            let mut filter = ImageFilter::crop(rect, None, None).unwrap();
226
227            let slr = if let Some(backdrop_filter) = backdrop_filter {
228                match backdrop_filter {
229                    Filter::Blur(radius) => {
230                        let sigma = radius.to_px().unwrap() * cx.scale_factor() / 2.0;
231                        filter = filter.blur(None, (sigma, sigma), None).unwrap();
232                        SaveLayerRec::default().paint(&paint).backdrop(&filter)
233                    }
234                }
235            } else {
236                SaveLayerRec::default().paint(&paint)
237            };
238
239            Some(canvas.save_layer(&slr))
240        } else {
241            None
242        };
243
244    canvas.save();
245    if let Some(Some(clip_path)) = cx.cache.clip_path.get(current) {
246        canvas.clip_path(clip_path, ClipOp::Intersect, true);
247    }
248
249    if let Some(transform) = cx.cache.transform.get(current) {
250        canvas.set_matrix(&(transform.into()));
251    }
252
253    let is_visible = match (visible, cx.visibility()) {
254        (v, None) => v,
255        (_, Some(Visibility::Hidden)) => false,
256        (_, Some(Visibility::Visible)) => true,
257    };
258
259    // Draw the view
260    if is_visible {
261        let should_draw = if let Some(dirty_rect) = dirty_rect {
262            let bounds = cached_draw_bounds(cx, current);
263            bounds.intersects(dirty_rect)
264        } else {
265            true
266        };
267
268        if should_draw {
269            if let Some(view) = cx.views.remove(&current) {
270                view.draw(cx, canvas);
271                cx.views.insert(current, view);
272            }
273        }
274    }
275    canvas.restore();
276    let child_iter = DrawChildIterator::new(cx.tree, cx.current);
277
278    // Draw its children
279    for child in child_iter {
280        cx.current = child;
281        // TODO: Skip views with zero-sized bounding boxes here? Or let user decide if they want to skip?
282        draw_entity(cx, dirty_rect, canvas, current_z, queue, is_visible);
283    }
284
285    if let Some(count) = layer_count {
286        canvas.restore_to_count(count);
287    }
288
289    cx.current = current;
290}

292fn cached_draw_bounds(cx: &mut DrawContext, entity: Entity) -> BoundingBox {
293    if let Some(bounds) = cx.cache.draw_bounds.get(entity).copied() {
294        return bounds;
295    }
296
297    let bounds = draw_bounds(cx.style, cx.cache, cx.tree, entity);
298    cx.cache.draw_bounds.insert(entity, bounds);
299    bounds
300}

302// Must be called after transform and clipping systems to be valid.
303pub(crate) fn draw_bounds(
304    style: &Style,
305    cache: &CachedData,
306    tree: &Tree<Entity>,
307    entity: Entity,
308) -> BoundingBox {
309    let mut layout_bounds = cache.bounds.get(entity).copied().unwrap();
310
311    if let Some(shadows) = style.shadow.get(entity) {
312        let original_bounds = layout_bounds;
313
314        for shadow in shadows.iter().filter(|shadow| !shadow.inset) {
315            let mut shadow_bounds = original_bounds;
316
317            let scale_factor = style.scale_factor();
318
319            let x = shadow.x_offset.to_px().unwrap() * scale_factor;
320            let y = shadow.y_offset.to_px().unwrap() * scale_factor;
321
322            shadow_bounds = shadow_bounds.offset(x, y);
323
324            if let Some(blur_radius) = shadow.blur_radius.as_ref().and_then(Length::to_px) {
325                shadow_bounds = shadow_bounds.expand(blur_radius * scale_factor);
326            }
327
328            if let Some(spread_radius) = shadow.spread_radius.as_ref().and_then(Length::to_px) {
329                shadow_bounds = shadow_bounds.expand(spread_radius * scale_factor);
330            }
331
332            layout_bounds = layout_bounds.union(&shadow_bounds);
333        }
334    }
335
336    let mut outline_bounds = layout_bounds;
337
338    if let Some(outline_width) = style.outline_width.get(entity) {
339        outline_bounds = outline_bounds
340            .expand(outline_width.to_pixels(layout_bounds.diagonal(), style.scale_factor()));
341    }
342
343    if let Some(outline_offset) = style.outline_offset.get(entity) {
344        outline_bounds = outline_bounds
345            .expand(outline_offset.to_pixels(layout_bounds.diagonal(), style.scale_factor()));
346    }
347
348    layout_bounds = layout_bounds.union(&outline_bounds);
349
350    let matrix = cache.transform.get(entity).copied().unwrap_or_default();
351
352    let (rect, _) = matrix.map_rect(Rect::from(layout_bounds));
353
354    let mut dirty_bounds = BoundingBox::from_min_max(
355        rect.left().floor(),
356        rect.top().floor(),
357        rect.right().ceil(),
358        rect.bottom().ceil(),
359    );
360
361    // If overflow is visible we have to include all children since they may draw outside of our own bounds.
362    if !matches!(style.overflowx.get(entity), Some(Overflow::Hidden))
363        || !matches!(style.overflowy.get(entity), Some(Overflow::Hidden))
364    {
365        let child_iter = DrawChildIterator::new(tree, entity);
366        for child in child_iter {
367            dirty_bounds = dirty_bounds.union(&draw_bounds(style, cache, tree, child));
368        }
369    }
370
371    // If z-index is not zero we can't be sure of the stacking context so we have to assume so and not clip.
372    if matches!(style.z_index.get(entity), Some(z_index) if *z_index != 0) {
373        return dirty_bounds;
374    }
375
376    if tree.is_window(entity) {
377        return dirty_bounds;
378    }
379
380    let parent = tree
381        .get_layout_parent(entity)
382        .unwrap_or_else(|| tree.get_parent_window(entity).unwrap_or(Entity::root()));
383
384    let Some(Some(clip_path)) = cache.clip_path.get(parent) else {
385        return dirty_bounds;
386    };
387
388    let clip_bounds = clip_path.bounds();
389
390    let clip_bounds = BoundingBox::from_min_max(
391        clip_bounds.left().floor(),
392        clip_bounds.top().floor(),
393        clip_bounds.right().ceil(),
394        clip_bounds.bottom().ceil(),
395    );
396
397    dirty_bounds.intersection(&clip_bounds)
398}

400struct ZEntity {
401    pub index: i32,
402    pub entity: Entity,
403    pub visible: bool,
404}
405
406impl Ord for ZEntity {
407    fn cmp(&self, other: &Self) -> Ordering {
408        other.index.cmp(&self.index)
409    }
410}
411impl PartialOrd for ZEntity {
412    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
413        Some(self.cmp(other))
414    }
415}
416impl PartialEq for ZEntity {
417    fn eq(&self, other: &Self) -> bool {
418        self.index == other.index
419    }
420}
