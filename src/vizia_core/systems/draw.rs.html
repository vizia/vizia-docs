20pub(crate) fn draw_system(
21    cx: &mut Context,
22    window_entity: Entity,
23    surface: &mut Surface,
24    dirty_surface: &mut Surface,
25) -> bool {
26    let _scope = crate::profiling::scope("draw_system");
27    if cx.windows.is_empty() {
28        return false;
29    }
30
31    if !cx.entity_manager.is_alive(window_entity) {
32        return false;
33    }
34
35    let window = cx.windows.get_mut(&window_entity).unwrap();
36
37    let mut dirty_rect = std::mem::take(&mut window.dirty_rect);
38    let redraw_list = std::mem::take(&mut window.redraw_list);
39
40    // Note: dirty_rect can be populated by entity removal independently of
41    // redraw_list, and backdrop filter processing generates dirty areas outside
42    // the redraw_list. Un-commenting would cause rendering artifacts. Maybe we
43    // can revisit this in the future.
44    // if redraw_list.is_empty() {
45    //     return false;
46    // }
47
48    let redraw_entities: Vec<Entity> = redraw_list
49        .iter()
50        .copied()
51        .filter(|&entity| {
52            !cx.tree.is_ignored(entity) && (!cx.tree.is_window(entity) || entity == window_entity)
53        })
54        .collect();
55
56    // The cached draw bounds of an entity cover all of its descendants, so a redrawn entity with a
57    // visible redrawn ancestor which has cached bounds does not need to dirty its own areas. The
58    // previous areas are computed before any cached bounds are invalidated.
59    let redraw_set: HashSet<Entity> = redraw_entities.iter().copied().collect();
60    let previous_bounds: Vec<(Entity, Option<BoundingBox>)> = redraw_entities
61        .iter()
62        .copied()
63        .filter(|&entity| {
64            !LayoutParentIterator::new(&cx.tree, entity).skip(1).any(|ancestor| {
65                redraw_set.contains(&ancestor)
66                    && cx.cache.draw_bounds.contains(ancestor)
67                    && ancestor.visible(&cx.style)
68            })
69        })
70        .map(|entity| (entity, cached_screen_bounds(&cx.cache, &cx.tree, entity)))
71        .collect();
72
73    for &entity in &redraw_entities {
74        cx.cache.invalidate_draw_bounds(&cx.tree, entity);
75    }
76
77    for (entity, previous) in previous_bounds {
78        // If the entity was previously visible but is no longer, its previous area must still be
79        // dirtied so that it is correctly cleared from the screen.
80        if let Some(previous) = previous {
81            union_dirty_rect(&mut dirty_rect, previous);
82        }
83
84        if entity.visible(&cx.style) {
85            let bounds = screen_bounds(&cx.style, &mut cx.cache, &cx.tree, entity);
86            union_dirty_rect(&mut dirty_rect, bounds);
87        }
88    }
89
90    for entity in cx.style.filter_entities(&cx.tree) {
91        if cx.tree.is_ignored(entity) {
92            continue;
93        }
94
95        if cx.tree.is_window(entity) && entity != window_entity {
96            continue;
97        }
98
99        if entity.visible(&cx.style) {
100            // Entity is VISIBLE and has a filter.
101            // Skip recomputation if already processed in redraw_list.
102            if !redraw_list.contains(&entity) {
103                let filter_current_bounds =
104                    screen_bounds(&cx.style, &mut cx.cache, &cx.tree, entity);
105
106                if is_non_empty(&filter_current_bounds) {
107                    // Condition to update dirty_rect:
108                    // 1. dirty_rect is None (then set it to filter_current_bounds).
109                    // 2. dirty_rect is Some, and filter_current_bounds intersects with it (then union).
110                    if dirty_rect.is_none_or(|current_dr_val| {
111                        filter_current_bounds.intersects(&current_dr_val)
112                    }) {
113                        dirty_rect =
114                            Some(dirty_rect.map_or(filter_current_bounds, |current_dr_val| {
115                                current_dr_val.union(&filter_current_bounds)
116                            }));
117                    }
118                }
119            }
120        } else {
121            // Entity is INVISIBLE but has (or had) a filter style.
122            // Its *previous* bounds need to be added to dirty_rect.
123            if let Some(previous_draw_bounds) = cached_screen_bounds(&cx.cache, &cx.tree, entity) {
124                union_dirty_rect(&mut dirty_rect, previous_draw_bounds);
125                // Remove from cache as it's no longer visible with these bounds.
126                cx.cache.invalidate_draw_bounds(&cx.tree, entity);
127            }
128        }
129    }
130
131    // if dirty_rect.is_none() {
132    //     return true;
133    // }
134
135    let canvas = dirty_surface.canvas();
136
137    canvas.save();
138
139    if let Some(rect) = dirty_rect.map(Rect::from) {
140        canvas.clip_rect(rect, ClipOp::Intersect, false);
141        canvas.clear(Color::transparent());
142    }
143
144    cx.resource_manager.mark_images_unused();
145
146    let mut heaps = ZHeaps::default();
147    let mut order = ZOrder::default();
148    let mut queue = ZQueue::new(&mut heaps, &mut order);
149    // With no dirty area nothing visible has changed, so the retained dirty surface is still
150    // valid and only needs compositing. Surface recreation (e.g. on resize) always dirties the
151    // whole window via `needs_refresh` or `needs_resize_redraw`.
152    if dirty_rect.is_some() {
153        queue.defer_raised(
154            0,
155            window_entity,
156            DeferredDraw {
157                visible: true,
158                ancestor_opacity: 1.0,
159                parent: ParentSpace::new(Matrix::new_identity(), None, &dirty_rect),
160            },
161        );
162    }
163
164    while let Some(zentity) = queue.pop(true) {
165        let count = save_ancestor_opacity(canvas, zentity.state.ancestor_opacity);
166        draw_entity(
167            &mut DrawContext {
168                current: zentity.entity,
169                style: &cx.style,
170                cache: &mut cx.cache,
171                tree: &cx.tree,
172                models: &cx.models,
173                views: &mut cx.views,
174                resource_manager: &cx.resource_manager,
175                text_context: &mut cx.text_context,
176                modifiers: &cx.modifiers,
177                mouse: &cx.mouse,
178                windows: &mut cx.windows,
179            },
180            &dirty_rect,
181            canvas,
182            zentity.index,
183            &mut queue,
184            zentity.state.visible,
185            false,
186            zentity.state.ancestor_opacity,
187            1.0,
188            &zentity.state.parent,
189        );
190        canvas.restore_to_count(count);
191    }
192
193    canvas.restore();
194
195    // Composite dirty surface with Src blend so transparent pixels in dirty
196    // regions replace destination pixels instead of preserving stale content.
197    let mut composite_paint = Paint::default();
198    composite_paint.set_blend_mode(skia_safe::BlendMode::Src);
199    dirty_surface.draw(
200        surface.canvas(),
201        (0, 0),
202        SamplingOptions::default(),
203        Some(&composite_paint),
204    );
205
206    // Debug overlay: outline the views which underwent layout in the most recent layout pass by
207    // stroking their transformed bounds with an orange rectangle. The overlay persists across
208    // frames (it is redrawn in full on every draw) until the layout system runs again, at which
209    // point the layout system repopulates `laid_out` and schedules the previous outlines to be
210    // erased.
211    if cx.style.debug_layout && !cx.style.laid_out.is_empty() {
212        let canvas = surface.canvas();
213        let mut paint = Paint::default();
214        paint.set_anti_alias(true);
215        paint.set_style(skia_safe::PaintStyle::Stroke);
216        paint.set_stroke_width(1.0);
217        // Orange (RGB 255, 165, 0).
218        paint.set_color(skia_safe::Color::from_argb(255, 255, 165, 0));
219
220        for entity in cx.style.laid_out.iter().copied() {
221            let Some(bounds) = cx.cache.bounds.get(entity).copied() else {
222                continue;
223            };
224            if bounds.w <= 0.0 || bounds.h <= 0.0 {
225                continue;
226            }
227            let matrix = cx.cache.absolute_transform(&cx.tree, entity);
228            let (rect, _) = matrix.map_rect(Rect::from(bounds));
229            canvas.draw_rect(rect, &paint);
230        }
231    }
232
233    // Debug draw dirty rect
234    // if let Some(rect) = dirty_rect.map(Rect::from) {
235    //     let mut paint = Paint::default();
236    //     paint.set_style(skia_safe::PaintStyle::Stroke);
237    //     paint.set_color(Color::red());
238    //     paint.set_stroke_width(1.0);
239    //     surface.canvas().draw_rect(rect, &paint);
240    // }
241
242    true
243}

245/// Saves the canvas and, for a queued view drawn outside some of its ancestors' opacity layers,
246/// re-enters the product of their opacities, as CSS applies an ancestor's opacity to every
247/// descendant. Returns the count to restore to once the view is drawn.
248fn save_ancestor_opacity(canvas: &Canvas, ancestor_opacity: f32) -> usize {
249    let count = canvas.save();
250    if ancestor_opacity != 1.0 {
251        let mut paint = Paint::default();
252        paint.set_alpha_f(ancestor_opacity);
253        canvas.save_layer(&SaveLayerRec::default().paint(&paint));
254    }
255    count
256}

258fn union_dirty_rect(dirty_rect: &mut Option<BoundingBox>, bounds: BoundingBox) {
259    if bounds.w <= 0.0 || bounds.h <= 0.0 {
260        return;
261    }
262
263    if let Some(current) = dirty_rect.as_mut() {
264        *current = current.union(&bounds);
265    } else {
266        *dirty_rect = Some(bounds);
267    }
268}

270#[allow(clippy::too_many_arguments)]
271fn draw_entity(
272    cx: &mut DrawContext,
273    dirty_rect: &Option<BoundingBox>,
274    canvas: &Canvas,
275    current_z: i32,
276    queue: &mut ZQueue<'_, DeferredDraw>,
277    visible: bool,
278    ancestor_backdrop_clip: bool,
279    ancestor_opacity: f32,
280    escape_opacity: f32,
281    parent: &ParentSpace,
282) {
283    let current = cx.current;
284
285    // Skip views with display: none.
286    if cx.display() == Display::None {
287        return;
288    }
289
290    // Skip the whole subtree if nothing within it can intersect the dirty rect. Subtree bounds
291    // include z-indexed and clip-escaping descendants, so it is safe to cull before they would be
292    // deferred to the z-order queue.
293    let draw_bounds = dirty_rect.map(|_| cached_bounds(cx.style, cx.cache, cx.tree, current));
294    if let (Some(dirty_rect), Some(bounds)) = (dirty_rect, &draw_bounds)
295        && !parent.intersects_dirty(bounds, dirty_rect)
296    {
297        return;
298    }
299
300    match Stacking::of(cx.style, current, current_z, ancestor_backdrop_clip) {
301        Stacking::Inline => {}
302        Stacking::Raised(index) => {
303            queue.defer_raised(
304                index,
305                current,
306                DeferredDraw { visible, ancestor_opacity, parent: *parent },
307            );
308            return;
309        }
310        Stacking::ClippingEscape(index) => {
311            // Clip escapes drain with their owner, outside the layers opened below it.
312            let ancestor_opacity = escape_opacity * ancestor_opacity;
313            queue.defer_clipping_escape(
314                index,
315                current,
316                DeferredDraw { visible, ancestor_opacity, parent: *parent },
317            );
318            return;
319        }
320    }
321
322    let transform = match cx.cache.local_transform.get(current) {
323        Some(local) if !is_identity(local) => parent.transform * *local,
324        _ => parent.transform,
325    };
326
327    let filter = cx.filter();
328    let backdrop_filter = cx.backdrop_filter();
329    let blend_mode = cx.style.blend_mode.get(current).copied().unwrap_or_default();
330
331    // Backdrop layers must be created inside the entity clip. Creating the
332    // layer first allows image-filter output to bleed past rounded clips.
333    let backdrop_clip_count = backdrop_filter.map(|_| {
334        let count = canvas.save();
335        if let Some(Some(clip_path)) = cx.cache.clip_path.get(current) {
336            canvas.clip_path(clip_path, ClipOp::Intersect, true);
337        }
338        count
339    });
340
341    let layer_count = if cx.opacity() != 1.0
342        || filter.is_some()
343        || backdrop_filter.is_some()
344        || blend_mode != BlendMode::Normal
345    {
346        let mut paint = Paint::default();
347        paint.set_alpha_f(cx.opacity());
348        paint.set_blend_mode(blend_mode.into());
349
350        let rect: Rect = cx.bounds().into();
351        let transformed_rect = transform.map_rect(rect).0;
352        let mut backdrop_image_filter = ImageFilter::crop(transformed_rect, None, None).unwrap();
353
354        if let Some(filter) = filter {
355            match filter {
356                Filter::Blur(radius) => {
357                    let sigma = radius.to_px().unwrap() * cx.scale_factor() / 2.0;
358                    let image_filter = ImageFilter::crop(rect, None, None)
359                        .unwrap()
360                        .blur(None, (sigma, sigma), None)
361                        .unwrap();
362                    paint.set_image_filter(image_filter);
363                }
364            }
365        }
366
367        let slr = if let Some(backdrop_filter) = backdrop_filter {
368            match backdrop_filter {
369                Filter::Blur(radius) => {
370                    let sigma = radius.to_px().unwrap() * cx.scale_factor() / 2.0;
371                    backdrop_image_filter =
372                        backdrop_image_filter.blur(None, (sigma, sigma), None).unwrap();
373                    SaveLayerRec::default()
374                        .bounds(&transformed_rect)
375                        .paint(&paint)
376                        .backdrop(&backdrop_image_filter)
377                }
378            }
379        } else {
380            SaveLayerRec::default().paint(&paint)
381        };
382
383        Some(canvas.save_layer(&slr))
384    } else {
385        None
386    };
387
388    let is_visible = match (visible, cx.visibility()) {
389        (v, None) => v,
390        (_, Some(Visibility::Hidden)) => false,
391        (_, Some(Visibility::Visible)) => true,
392    };
393
394    let should_draw = if is_visible {
395        if let (Some(dirty_rect), Some(bounds)) = (dirty_rect, &draw_bounds) {
396            intersects_non_empty(&own_to_screen(bounds, &parent.transform, parent.clip), dirty_rect)
397        } else {
398            true
399        }
400    } else {
401        false
402    };
403
404    // Draw the view. Children set up their own clip and absolute transform, so the canvas state
405    // only needs to be configured when this view actually paints.
406    if should_draw {
407        canvas.save();
408        if backdrop_clip_count.is_none()
409            && let Some(Some(clip_path)) = cx.cache.clip_path.get(current)
410        {
411            canvas.clip_path(clip_path, ClipOp::Intersect, true);
412        }
413        canvas.set_matrix(&(transform.into()));
414
415        if let Some(view) = cx.views.remove(&current) {
416            view.draw(cx, canvas);
417            cx.views.insert(current, view);
418        }
419        canvas.restore();
420    }
421    let child_iter = DrawChildIterator::new(cx.tree, cx.current);
422    let child_space = ParentSpace::new(transform, clip_bounds(cx.cache, current), dirty_rect);
423    let opacity = ancestor_opacity * cx.opacity();
424
425    // An opacity group composites its complete image once, including raised descendants and the
426    // clip escapes it owns, so they inherit its opacity from its open layer.
427    let group = StackingGroup::of(cx.style, current, ancestor_backdrop_clip);
428    let (descendants_ancestor_opacity, descendants_escape_opacity) = if group.owns_raised {
429        (1.0, if group.owns_escapes { 1.0 } else { escape_opacity * opacity })
430    } else {
431        (opacity, escape_opacity)
432    };
433    let mut group_heaps = queue.group_heaps(group);
434    let mut child_queue = queue.descend(group, &mut group_heaps);
435
436    for child in child_iter {
437        cx.current = child;
438        // TODO: Skip views with zero-sized bounding boxes here? Or let user decide if they want to skip?
439        draw_entity(
440            cx,
441            dirty_rect,
442            canvas,
443            current_z,
444            &mut child_queue,
445            is_visible,
446            group.backdrop_clipped,
447            descendants_ancestor_opacity,
448            descendants_escape_opacity,
449            &child_space,
450        );
451    }
452
453    if group.owns_raised {
454        while let Some(zentity) = child_queue.pop(group.owns_escapes) {
455            cx.current = zentity.entity;
456            // This group's layer is still open, but layers between it and the view have closed.
457            let count = save_ancestor_opacity(canvas, zentity.state.ancestor_opacity);
458            draw_entity(
459                cx,
460                dirty_rect,
461                canvas,
462                zentity.index,
463                &mut child_queue,
464                zentity.state.visible,
465                group.backdrop_clipped,
466                zentity.state.ancestor_opacity,
467                descendants_escape_opacity,
468                &zentity.state.parent,
469            );
470            canvas.restore_to_count(count);
471        }
472    }
473    queue.recycle(group_heaps);
474
475    cx.current = current;
476
477    // Draw outline after children so it appears above descendant content.
478    if should_draw {
479        canvas.save();
480        if let Some(Some(clip_path)) = cx.cache.clip_path.get(current) {
481            canvas.clip_path(clip_path, ClipOp::Intersect, true);
482        }
483
484        canvas.set_matrix(&(transform.into()));
485
486        cx.draw_outline(canvas);
487        canvas.restore();
488    }
489
490    if let Some(count) = layer_count {
491        canvas.restore_to_count(count);
492    }
493    if let Some(count) = backdrop_clip_count {
494        canvas.restore_to_count(count);
495    }
496}

498fn is_non_empty(bounds: &BoundingBox) -> bool {
499    bounds.w > 0.0 && bounds.h > 0.0

502fn intersects_non_empty(a: &BoundingBox, b: &BoundingBox) -> bool {
503    is_non_empty(a) && is_non_empty(b) && a.intersects(b)
504}

506fn union_non_empty(a: BoundingBox, b: BoundingBox) -> BoundingBox {
507    match (is_non_empty(&a), is_non_empty(&b)) {
508        (true, true) => a.union(&b),
509        (true, false) => a,
510        _ => b,
511    }
512}

514fn snap(rect: Rect) -> BoundingBox {
515    BoundingBox::from_min_max(
516        rect.left().floor(),
517        rect.top().floor(),
518        rect.right().ceil(),
519        rect.bottom().ceil(),
520    )

523/// Maps bounds through a transform, expanding the result to whole pixels.
524fn map_bounds(transform: &Matrix, bounds: BoundingBox) -> BoundingBox {
525    if !is_non_empty(&bounds) {
526        return BoundingBox::default();
527    }
528
529    // Fast path for scale and translation, which avoids a call into Skia for every culled entity.
530    if is_scale_translate(transform) {
531        let (sx, sy) = (transform.scale_x(), transform.scale_y());
532        let (tx, ty) = (transform.translate_x(), transform.translate_y());
533        let (x0, x1) = (bounds.left() * sx + tx, bounds.right() * sx + tx);
534        let (y0, y1) = (bounds.top() * sy + ty, bounds.bottom() * sy + ty);
535        return snap(Rect::new(x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1)));
536    }
537
538    snap(transform.map_rect(Rect::from(bounds)).0)
539}

546/// Converts the cached draw bounds of an entity and its descendants into window coordinates, given
547/// the transform and clip bounds of the parent of the entity in window coordinates.
548pub(crate) fn subtree_to_screen(
549    bounds: &DrawBounds,
550    parent_transform: &Matrix,
551    parent_clip: Option<BoundingBox>,
552) -> BoundingBox {
553    let clipped = map_bounds(parent_transform, bounds.clipped);
554    let clipped = match parent_clip {
555        Some(clip) if is_non_empty(&clipped) => clipped.intersection(&clip),
556        _ => clipped,
557    };
558    if !is_non_empty(&bounds.escaped) {
559        return clipped;
560    }
561    union_non_empty(clipped, map_bounds(parent_transform, bounds.escaped))
562}

564/// Converts the cached bounds drawn by an entity itself into window coordinates.
565fn own_to_screen(
566    bounds: &DrawBounds,
567    parent_transform: &Matrix,
568    parent_clip: Option<BoundingBox>,
569) -> BoundingBox {
570    let own = map_bounds(parent_transform, bounds.own);
571    match parent_clip {
572        Some(clip) if !bounds.escapes_clip && is_non_empty(&own) => own.intersection(&clip),
573        _ => own,
574    }
575}

577/// Returns the transform and clip bounds, in window coordinates, which apply to the cached draw
578/// bounds of an entity.
579fn parent_space(
580    cache: &CachedData,
581    tree: &Tree<Entity>,
582    entity: Entity,
583) -> (Matrix, Option<BoundingBox>) {
584    if tree.is_window(entity) {
585        return (Matrix::new_identity(), None);
586    }
587
588    let parent = tree
589        .get_layout_parent(entity)
590        .unwrap_or_else(|| tree.get_parent_window(entity).unwrap_or(Entity::root()));
591
592    (cache.absolute_transform(tree, parent), clip_bounds(cache, parent))
593}

630/// Returns the cached draw bounds of an entity, computing and caching them (and those of its
631/// descendants) if necessary.
632fn cached_bounds(
633    style: &Style,
634    cache: &mut CachedData,
635    tree: &Tree<Entity>,
636    entity: Entity,
637) -> DrawBounds {
638    if let Some(bounds) = cache.draw_bounds.get(entity).copied() {
639        return bounds;
640    }
641

643    cache.draw_bounds.insert(entity, bounds);
644    bounds

647/// Computes the draw bounds of an entity in the coordinate space of its parent.
648// Must be called after the layout and transform systems to be valid.
649fn draw_bounds(
650    style: &Style,
651    cache: &mut CachedData,
652    tree: &Tree<Entity>,
653    entity: Entity,
654) -> DrawBounds {
655    let mut own = decorated_bounds(style, cache, entity);
656
657    let mut clipped = BoundingBox::default();
658    let mut escaped = BoundingBox::default();
659    for child in DrawChildIterator::new(tree, entity) {
660        let child_bounds = cached_bounds(style, cache, tree, child);
661        clipped = union_non_empty(clipped, child_bounds.clipped);
662        escaped = union_non_empty(escaped, child_bounds.escaped);
663    }
664
665    if is_non_empty(&clipped)
666        && let Some(clip) = own_clip_bounds(style, cache, tree, entity)
667    {
668        clipped = clipped.intersection(&clip);
669    }
670    let mut clipped = union_non_empty(own, clipped);
671
672    if let Some(Filter::Blur(radius)) = style.filter.get(entity) {
673        let radius = radius.to_px().unwrap() * style.scale_factor();
674        own = own.expand(radius);
675        clipped = clipped.expand(radius);
676        if is_non_empty(&escaped) {
677            escaped = escaped.expand(radius);
678        }
679    }
680
681    let transform = cache.local_transform.get(entity).copied().unwrap_or_default();
682    let own = map_bounds(&transform, own);
683    let mut clipped = map_bounds(&transform, clipped);
684    let mut escaped = map_bounds(&transform, escaped);
685
686    // If z-index is not zero we can't be sure of the stacking context so we have to assume so and
687    // not clip.
688    let escapes_clip = tree.is_window(entity)
689        || !should_apply_entity_clip(style, entity)
690        || matches!(style.z_index.get(entity), Some(z_index) if *z_index != 0);
691
692    if escapes_clip {
693        escaped = union_non_empty(escaped, clipped);
694        clipped = BoundingBox::default();
695    }
696
697    DrawBounds { own, clipped, escaped, escapes_clip }
698}

700/// Returns the layout bounds of an entity expanded by its shadows and outline.
701fn decorated_bounds(style: &Style, cache: &CachedData, entity: Entity) -> BoundingBox {
702    let mut layout_bounds = cache.bounds.get(entity).copied().unwrap_or_default();
703    if !is_non_empty(&layout_bounds) {
704        return BoundingBox::default();
705    }
706
707    if let Some(shadows) = style.shadow.get_resolved(entity, &style.custom_shadow_props) {
708        let original_bounds = layout_bounds;
709
710        for shadow in shadows.iter().filter(|shadow| !shadow.inset) {
711            let mut shadow_bounds = original_bounds;
712
713            let scale_factor = style.scale_factor();
714
715            let x = shadow.x_offset.to_px().unwrap() * scale_factor;
716            let y = shadow.y_offset.to_px().unwrap() * scale_factor;
717
718            shadow_bounds = shadow_bounds.offset(x, y);
719
720            if let Some(blur_radius) = shadow.blur_radius.as_ref().and_then(Length::to_px) {
721                shadow_bounds = shadow_bounds.expand(blur_radius * scale_factor);
722            }
723
724            if let Some(spread_radius) = shadow.spread_radius.as_ref().and_then(Length::to_px) {
725                shadow_bounds = shadow_bounds.expand(spread_radius * scale_factor);
726            }
727
728            layout_bounds = layout_bounds.union(&shadow_bounds);
729        }
730    }
731
732    let mut outline_bounds = layout_bounds;
733
734    if let Some(outline_width) =
735        style.outline_width.get_resolved(entity, &style.custom_length_props)
736    {
737        outline_bounds = outline_bounds
738            .expand(outline_width.to_pixels(layout_bounds.diagonal(), style.scale_factor()));
739    }
740
741    if let Some(outline_offset) =
742        style.outline_offset.get_resolved(entity, &style.custom_length_props)
743    {
744        outline_bounds = outline_bounds
745            .expand(outline_offset.to_pixels(layout_bounds.diagonal(), style.scale_factor()));
746    }
747
748    layout_bounds.union(&outline_bounds)
749}

751/// The coordinate space of the parent of an entity being drawn.
752#[derive(Clone, Copy)]
753struct ParentSpace {
754    /// The transform of the parent in window coordinates.
755    transform: Matrix,
756    /// The clip bounds of the parent in window coordinates.
757    clip: Option<BoundingBox>,
758    /// The dirty rect in the coordinate space of the parent, if the parent transform only scales
759    /// and translates. Allows children to be culled using their cached draw bounds directly.
760    local_dirty: Option<LocalDirty>,
761}
762
763#[derive(Clone, Copy)]
764struct LocalDirty {
765    /// The dirty rect intersected with the parent clip, for comparison with clipped bounds.
766    clipped: BoundingBox,
767    /// The dirty rect, for comparison with bounds which escape the parent clip.
768    escaped: BoundingBox,
769}
770
771impl ParentSpace {
772    fn new(transform: Matrix, clip: Option<BoundingBox>, dirty_rect: &Option<BoundingBox>) -> Self {
773        let local_dirty = dirty_rect.and_then(|dirty| {
774            let (sx, sy) = (transform.scale_x(), transform.scale_y());
775            if !is_scale_translate(&transform) || sx == 0.0 || sy == 0.0 {
776                return None;
777            }
778            let (tx, ty) = (transform.translate_x(), transform.translate_y());
779            let to_local = |bounds: BoundingBox| {
780                if !is_non_empty(&bounds) {
781                    return BoundingBox::default();
782                }
783                // Window space bounds are expanded to whole pixels, so expand the dirty rect to
784                // keep culling conservative.
785                let bounds = bounds.expand(1.0);
786                let (x0, x1) = ((bounds.left() - tx) / sx, (bounds.right() - tx) / sx);
787                let (y0, y1) = ((bounds.top() - ty) / sy, (bounds.bottom() - ty) / sy);
788                BoundingBox::from_min_max(x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1))
789            };
790            let clipped = clip.map_or(dirty, |clip| dirty.intersection(&clip));
791            Some(LocalDirty { clipped: to_local(clipped), escaped: to_local(dirty) })
792        });
793        Self { transform, clip, local_dirty }
794    }
795
796    /// Returns true if the drawn bounds of an entity and its descendants may intersect the dirty
797    /// rect.
798    fn intersects_dirty(&self, bounds: &DrawBounds, dirty_rect: &BoundingBox) -> bool {
799        match &self.local_dirty {
800            Some(local) => {
801                intersects_non_empty(&bounds.clipped, &local.clipped)
802                    || intersects_non_empty(&bounds.escaped, &local.escaped)
803            }
804            None => intersects_non_empty(
805                &subtree_to_screen(bounds, &self.transform, self.clip),
806                dirty_rect,
807            ),
808        }
809    }
810}

820/// Where the stacking order places a view reached at `current_z`. Drawing and hit testing both
821/// follow it, so the topmost view hit is the topmost view painted.
822pub(crate) enum Stacking {
823    /// The view is visited with its parent, in tree order.
824    Inline,
825    /// The view is deferred to the raised queue at its z-index.
826    Raised(i32),
827    /// The view is deferred to the clip escape queue, which its owner drains outside the backdrop
828    /// clips between them.
829    ClippingEscape(i32),
830}
831
832impl Stacking {
833    pub(crate) fn of(
834        style: &Style,
835        entity: Entity,
836        current_z: i32,
837        ancestor_backdrop_clip: bool,
838    ) -> Self {
839        let z_index = style.z_index.get(entity).copied().unwrap_or_default();
840
841        // A Skia clip cannot be expanded until its saved canvas state is restored. Defer views
842        // which explicitly escape ancestor clipping so a backdrop clip higher in the recursive draw
843        // cannot accidentally constrain popovers and other overlays.
844        if ancestor_backdrop_clip && !should_apply_entity_clip(style, entity) {
845            Stacking::ClippingEscape(current_z.max(z_index))
846        } else if z_index > current_z {
847            Stacking::Raised(z_index)
848        } else {
849            Stacking::Inline
850        }
851    }
852}

854/// The queues the descendants of a view visited inline are deferred to.
855#[derive(Clone, Copy)]
856pub(crate) struct StackingGroup {
857    /// The view is an opacity group, which composites its raised descendants itself.
858    pub owns_raised: bool,
859    /// The view also owns the clip escapes below it. A Skia clip cannot be widened while it is
860    /// active, so only a group outside every backdrop clip owns them; a group inside one leaves
861    /// them to an outer owner.
862    pub owns_escapes: bool,
863    /// The descendants are drawn inside a backdrop clip.
864    pub backdrop_clipped: bool,
865}
866
867impl StackingGroup {
868    pub(crate) fn of(style: &Style, entity: Entity, ancestor_backdrop_clip: bool) -> Self {
869        let opacity = style
870            .opacity
871            .get_resolved(entity, &style.custom_opacity_props)
872            .unwrap_or(Opacity(1.0))
873            .0;
874        let backdrop_filter = style.backdrop_filter.get(entity).is_some();
875        let owns_raised = opacity != 1.0
876            && style.filter.get(entity).is_none()
877            && !backdrop_filter
878            && style.blend_mode.get(entity).copied().unwrap_or_default() == BlendMode::Normal;
879
880        StackingGroup {
881            owns_raised,
882            owns_escapes: owns_raised && !ancestor_backdrop_clip,
883            backdrop_clipped: ancestor_backdrop_clip || backdrop_filter,
884        }
885    }
886}

896/// The heaps of the z-order queues owned by a window or an opacity group.
897pub(crate) struct ZHeaps<T> {
898    raised: BinaryHeap<ZEntity<T>>,
899    clipping_escapes: BinaryHeap<ZEntity<T>>,
900}
901
902impl<T> Default for ZHeaps<T> {
903    fn default() -> Self {
904        Self { raised: BinaryHeap::new(), clipping_escapes: BinaryHeap::new() }
905    }
906}

908/// The state shared by every z-order queue of one traversal.
909pub(crate) struct ZOrder<T> {
910    /// The tree order of the next deferred view, which orders views with equal z-indices.
911    next_order: usize,
912    /// Drained group heaps, which later groups reuse instead of allocating their own.
913    spare: Vec<ZHeaps<T>>,
914}
915
916impl<T> Default for ZOrder<T> {
917    fn default() -> Self {
918        Self { next_order: 0, spare: Vec::new() }
919    }
920}
921
922/// The z-order queues a view defers its raised descendants and clip escapes to.
923pub(crate) struct ZQueue<'a, T> {
924    raised: &'a mut BinaryHeap<ZEntity<T>>,
925    clipping_escapes: &'a mut BinaryHeap<ZEntity<T>>,
926    order: &'a mut ZOrder<T>,
927}
928
929impl<'a, T> ZQueue<'a, T> {
930    pub(crate) fn new(heaps: &'a mut ZHeaps<T>, order: &'a mut ZOrder<T>) -> Self {
931        Self { raised: &mut heaps.raised, clipping_escapes: &mut heaps.clipping_escapes, order }
932    }
933
934    fn entry(&mut self, index: i32, entity: Entity, state: T) -> ZEntity<T> {
935        let order = self.order.next_order;
936        self.order.next_order += 1;
937        ZEntity { index, entity, order, state }
938    }
939
940    /// Returns the heaps for the queues a view in `group` owns, reusing those of a drained group.
941    pub(crate) fn group_heaps(&mut self, group: StackingGroup) -> ZHeaps<T> {
942        if group.owns_raised {
943            self.order.spare.pop().unwrap_or_default()
944        } else {
945            ZHeaps::default()
946        }
947    }
948
949    /// Keeps the heaps of a drained group for reuse.
950    pub(crate) fn recycle(&mut self, heaps: ZHeaps<T>) {
951        debug_assert!(heaps.raised.is_empty() && heaps.clipping_escapes.is_empty());
952        if heaps.raised.capacity() > 0 || heaps.clipping_escapes.capacity() > 0 {
953            self.order.spare.push(heaps);
954        }
955    }
956
957    pub(crate) fn defer_raised(&mut self, index: i32, entity: Entity, state: T) {
958        let entry = self.entry(index, entity, state);
959        self.raised.push(entry);
960    }
961
962    pub(crate) fn defer_clipping_escape(&mut self, index: i32, entity: Entity, state: T) {
963        let entry = self.entry(index, entity, state);
964        self.clipping_escapes.push(entry);
965    }
966
967    /// Returns the queues the descendants of a view in `group` are deferred to, taking those the
968    /// view owns from `heaps`.
969    pub(crate) fn descend<'b>(
970        &'b mut self,
971        group: StackingGroup,
972        heaps: &'b mut ZHeaps<T>,
973    ) -> ZQueue<'b, T> {
974        ZQueue {
975            raised: if group.owns_raised { &mut heaps.raised } else { &mut *self.raised },
976            clipping_escapes: if group.owns_escapes {
977                &mut heaps.clipping_escapes
978            } else {
979                &mut *self.clipping_escapes
980            },
981            order: &mut *self.order,
982        }
983    }
984
985    /// Pops the next view for the owner of the raised queue to visit, including the clip escapes
986    /// if it owns them too.
987    pub(crate) fn pop(&mut self, owns_escapes: bool) -> Option<ZEntity<T>> {
988        if owns_escapes {
989            self.raised.append(self.clipping_escapes);
990        }
991        self.raised.pop()
992    }
993}

995impl<T> Ord for ZEntity<T> {
996    fn cmp(&self, other: &Self) -> Ordering {
997        other.index.cmp(&self.index).then_with(|| other.order.cmp(&self.order))
998    }
999}
1000impl<T> PartialOrd for ZEntity<T> {
1001    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
1002        Some(self.cmp(other))
1003    }
1004}
1005impl<T> PartialEq for ZEntity<T> {
1006    fn eq(&self, other: &Self) -> bool {
1007        self.index == other.index && self.order == other.order
1008    }
1009}
