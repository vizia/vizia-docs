20pub(crate) fn draw_system(
21    cx: &mut Context,
22    window_entity: Entity,
23    surface: &mut Surface,
24    dirty_surface: &mut Surface,
25) -> bool {
26    let _scope = crate::profiling::scope("draw_system");
27    if cx.windows.is_empty() {
28        return false;
29    }
30
31    if !cx.entity_manager.is_alive(window_entity) {
32        return false;
33    }
34
35    let window = cx.windows.get_mut(&window_entity).unwrap();
36
37    let mut dirty_rect = std::mem::take(&mut window.dirty_rect);
38    let redraw_list = std::mem::take(&mut window.redraw_list);
39
40    // Note: dirty_rect can be populated by entity removal independently of
41    // redraw_list, and backdrop filter processing generates dirty areas outside
42    // the redraw_list. Un-commenting would cause rendering artifacts. Maybe we
43    // can revisit this in the future.
44    // if redraw_list.is_empty() {
45    //     return false;
46    // }
47
48    let redraw_entities: Vec<Entity> = redraw_list
49        .iter()
50        .copied()
51        .filter(|&entity| {
52            !cx.tree.is_ignored(entity) && (!cx.tree.is_window(entity) || entity == window_entity)
53        })
54        .collect();
55
56    // The cached draw bounds of an entity cover all of its descendants, so a redrawn entity with a
57    // visible redrawn ancestor which has cached bounds does not need to dirty its own areas. The
58    // previous areas are computed before any cached bounds are invalidated.
59    let redraw_set: HashSet<Entity> = redraw_entities.iter().copied().collect();
60    let previous_bounds: Vec<(Entity, Option<BoundingBox>)> = redraw_entities
61        .iter()
62        .copied()
63        .filter(|&entity| {
64            !LayoutParentIterator::new(&cx.tree, entity).skip(1).any(|ancestor| {
65                redraw_set.contains(&ancestor)
66                    && cx.cache.draw_bounds.contains(ancestor)
67                    && ancestor.visible(&cx.style)
68            })
69        })
70        .map(|entity| (entity, cached_screen_bounds(&cx.cache, &cx.tree, entity)))
71        .collect();
72
73    for &entity in &redraw_entities {
74        cx.cache.invalidate_draw_bounds(&cx.tree, entity);
75    }
76
77    for (entity, previous) in previous_bounds {
78        // If the entity was previously visible but is no longer, its previous area must still be
79        // dirtied so that it is correctly cleared from the screen.
80        if let Some(previous) = previous {
81            union_dirty_rect(&mut dirty_rect, previous);
82        }
83
84        if entity.visible(&cx.style) {
85            let bounds = screen_bounds(&cx.style, &mut cx.cache, &cx.tree, entity);
86            union_dirty_rect(&mut dirty_rect, bounds);
87        }
88    }
89
90    for entity in cx.style.filter_entities(&cx.tree) {
91        if cx.tree.is_ignored(entity) {
92            continue;
93        }
94
95        if cx.tree.is_window(entity) && entity != window_entity {
96            continue;
97        }
98
99        if entity.visible(&cx.style) {
100            // Entity is VISIBLE and has a filter.
101            // Skip recomputation if already processed in redraw_list.
102            if !redraw_list.contains(&entity) {
103                let filter_current_bounds =
104                    screen_bounds(&cx.style, &mut cx.cache, &cx.tree, entity);
105
106                if is_non_empty(&filter_current_bounds) {
107                    // Condition to update dirty_rect:
108                    // 1. dirty_rect is None (then set it to filter_current_bounds).
109                    // 2. dirty_rect is Some, and filter_current_bounds intersects with it (then union).
110                    if dirty_rect.is_none_or(|current_dr_val| {
111                        filter_current_bounds.intersects(&current_dr_val)
112                    }) {
113                        dirty_rect =
114                            Some(dirty_rect.map_or(filter_current_bounds, |current_dr_val| {
115                                current_dr_val.union(&filter_current_bounds)
116                            }));
117                    }
118                }
119            }
120        } else {
121            // Entity is INVISIBLE but has (or had) a filter style.
122            // Its *previous* bounds need to be added to dirty_rect.
123            if let Some(previous_draw_bounds) = cached_screen_bounds(&cx.cache, &cx.tree, entity) {
124                union_dirty_rect(&mut dirty_rect, previous_draw_bounds);
125                // Remove from cache as it's no longer visible with these bounds.
126                cx.cache.invalidate_draw_bounds(&cx.tree, entity);
127            }
128        }
129    }
130
131    // if dirty_rect.is_none() {
132    //     return true;
133    // }
134
135    let canvas = dirty_surface.canvas();
136
137    canvas.save();
138
139    if let Some(rect) = dirty_rect.map(Rect::from) {
140        canvas.clip_rect(rect, ClipOp::Intersect, false);
141        canvas.clear(Color::transparent());
142    }
143
144    cx.resource_manager.mark_images_unused();
145
146    let mut queue = BinaryHeap::new();
147    // With no dirty area nothing visible has changed, so the retained dirty surface is still
148    // valid and only needs compositing. Surface recreation (e.g. on resize) always dirties the
149    // whole window via `needs_refresh` or `needs_resize_redraw`.
150    if dirty_rect.is_some() {
151        queue.push(ZEntity {
152            index: 0,
153            entity: window_entity,
154            visible: true,
155            parent: ParentSpace::new(Matrix::new_identity(), None, &dirty_rect),
156        });
157    }
158
159    while let Some(zentity) = queue.pop() {
160        canvas.save();
161        draw_entity(
162            &mut DrawContext {
163                current: zentity.entity,
164                style: &cx.style,
165                cache: &mut cx.cache,
166                tree: &cx.tree,
167                models: &cx.models,
168                views: &mut cx.views,
169                resource_manager: &cx.resource_manager,
170                text_context: &mut cx.text_context,
171                modifiers: &cx.modifiers,
172                mouse: &cx.mouse,
173                windows: &mut cx.windows,
174            },
175            &dirty_rect,
176            canvas,
177            zentity.index,
178            &mut queue,
179            zentity.visible,
180            false,
181            &zentity.parent,
182        );
183        canvas.restore();
184    }
185
186    canvas.restore();
187
188    // Composite dirty surface with Src blend so transparent pixels in dirty
189    // regions replace destination pixels instead of preserving stale content.
190    let mut composite_paint = Paint::default();
191    composite_paint.set_blend_mode(skia_safe::BlendMode::Src);
192    dirty_surface.draw(
193        surface.canvas(),
194        (0, 0),
195        SamplingOptions::default(),
196        Some(&composite_paint),
197    );
198
199    // Debug overlay: outline the views which underwent layout in the most recent layout pass by
200    // stroking their transformed bounds with an orange rectangle. The overlay persists across
201    // frames (it is redrawn in full on every draw) until the layout system runs again, at which
202    // point the layout system repopulates `laid_out` and schedules the previous outlines to be
203    // erased.
204    if cx.style.debug_layout && !cx.style.laid_out.is_empty() {
205        let canvas = surface.canvas();
206        let mut paint = Paint::default();
207        paint.set_anti_alias(true);
208        paint.set_style(skia_safe::PaintStyle::Stroke);
209        paint.set_stroke_width(1.0);
210        // Orange (RGB 255, 165, 0).
211        paint.set_color(skia_safe::Color::from_argb(255, 255, 165, 0));
212
213        for entity in cx.style.laid_out.iter().copied() {
214            let Some(bounds) = cx.cache.bounds.get(entity).copied() else {
215                continue;
216            };
217            if bounds.w <= 0.0 || bounds.h <= 0.0 {
218                continue;
219            }
220            let matrix = cx.cache.absolute_transform(&cx.tree, entity);
221            let (rect, _) = matrix.map_rect(Rect::from(bounds));
222            canvas.draw_rect(rect, &paint);
223        }
224    }
225
226    // Debug draw dirty rect
227    // if let Some(rect) = dirty_rect.map(Rect::from) {
228    //     let mut paint = Paint::default();
229    //     paint.set_style(skia_safe::PaintStyle::Stroke);
230    //     paint.set_color(Color::red());
231    //     paint.set_stroke_width(1.0);
232    //     surface.canvas().draw_rect(rect, &paint);
233    // }
234
235    true
236}

238fn union_dirty_rect(dirty_rect: &mut Option<BoundingBox>, bounds: BoundingBox) {
239    if bounds.w <= 0.0 || bounds.h <= 0.0 {
240        return;
241    }
242
243    if let Some(current) = dirty_rect.as_mut() {
244        *current = current.union(&bounds);
245    } else {
246        *dirty_rect = Some(bounds);
247    }
248}

250#[allow(clippy::too_many_arguments)]
251fn draw_entity(
252    cx: &mut DrawContext,
253    dirty_rect: &Option<BoundingBox>,
254    canvas: &Canvas,
255    current_z: i32,
256    queue: &mut BinaryHeap<ZEntity>,
257    visible: bool,
258    ancestor_backdrop_clip: bool,
259    parent: &ParentSpace,
260) {
261    let current = cx.current;
262
263    // Skip views with display: none.
264    if cx.display() == Display::None {
265        return;
266    }
267
268    // Skip the whole subtree if nothing within it can intersect the dirty rect. Subtree bounds
269    // include z-indexed and clip-escaping descendants, so it is safe to cull before they would be
270    // deferred to the z-order queue.
271    let draw_bounds = dirty_rect.map(|_| cached_bounds(cx.style, cx.cache, cx.tree, current));
272    if let (Some(dirty_rect), Some(bounds)) = (dirty_rect, &draw_bounds) {
273        if !parent.intersects_dirty(bounds, dirty_rect) {
274            return;
275        }
276    }
277
278    let z_index = cx.z_index();
279
280    let defer = |queue: &mut BinaryHeap<ZEntity>, index: i32| {
281        queue.push(ZEntity { index, entity: current, visible, parent: *parent });
282    };
283
284    if z_index > current_z {
285        defer(queue, z_index);
286        return;
287    }
288
289    // A Skia clip cannot be expanded until its saved canvas state is restored. Defer views which
290    // explicitly escape ancestor clipping so a backdrop clip higher in the recursive draw cannot
291    // accidentally constrain popovers and other overlays.
292    if ancestor_backdrop_clip && !should_apply_entity_clip(cx.style, current) {
293        defer(queue, current_z.max(z_index));
294        return;
295    }
296
297    let transform = match cx.cache.local_transform.get(current) {
298        Some(local) if !is_identity(local) => parent.transform * *local,
299        _ => parent.transform,
300    };
301
302    let filter = cx.filter();
303    let backdrop_filter = cx.backdrop_filter();
304    let blend_mode = cx.style.blend_mode.get(current).copied().unwrap_or_default();
305
306    // Backdrop layers must be created inside the entity clip. Creating the
307    // layer first allows image-filter output to bleed past rounded clips.
308    let backdrop_clip_count = backdrop_filter.map(|_| {
309        let count = canvas.save();
310        if let Some(Some(clip_path)) = cx.cache.clip_path.get(current) {
311            canvas.clip_path(clip_path, ClipOp::Intersect, true);
312        }
313        count
314    });
315
316    let layer_count = if cx.opacity() != 1.0
317        || filter.is_some()
318        || backdrop_filter.is_some()
319        || blend_mode != BlendMode::Normal
320    {
321        let mut paint = Paint::default();
322        paint.set_alpha_f(cx.opacity());
323        paint.set_blend_mode(blend_mode.into());
324
325        let rect: Rect = cx.bounds().into();
326        let transformed_rect = transform.map_rect(rect).0;
327        let mut backdrop_image_filter = ImageFilter::crop(transformed_rect, None, None).unwrap();
328
329        if let Some(filter) = filter {
330            match filter {
331                Filter::Blur(radius) => {
332                    let sigma = radius.to_px().unwrap() * cx.scale_factor() / 2.0;
333                    let image_filter = ImageFilter::crop(rect, None, None)
334                        .unwrap()
335                        .blur(None, (sigma, sigma), None)
336                        .unwrap();
337                    paint.set_image_filter(image_filter);
338                }
339            }
340        }
341
342        let slr = if let Some(backdrop_filter) = backdrop_filter {
343            match backdrop_filter {
344                Filter::Blur(radius) => {
345                    let sigma = radius.to_px().unwrap() * cx.scale_factor() / 2.0;
346                    backdrop_image_filter =
347                        backdrop_image_filter.blur(None, (sigma, sigma), None).unwrap();
348                    SaveLayerRec::default()
349                        .bounds(&transformed_rect)
350                        .paint(&paint)
351                        .backdrop(&backdrop_image_filter)
352                }
353            }
354        } else {
355            SaveLayerRec::default().paint(&paint)
356        };
357
358        Some(canvas.save_layer(&slr))
359    } else {
360        None
361    };
362
363    let is_visible = match (visible, cx.visibility()) {
364        (v, None) => v,
365        (_, Some(Visibility::Hidden)) => false,
366        (_, Some(Visibility::Visible)) => true,
367    };
368
369    let should_draw = if is_visible {
370        if let (Some(dirty_rect), Some(bounds)) = (dirty_rect, &draw_bounds) {
371            intersects_non_empty(&own_to_screen(bounds, &parent.transform, parent.clip), dirty_rect)
372        } else {
373            true
374        }
375    } else {
376        false
377    };
378
379    // Draw the view. Children set up their own clip and absolute transform, so the canvas state
380    // only needs to be configured when this view actually paints.
381    if should_draw {
382        canvas.save();
383        if backdrop_clip_count.is_none() {
384            if let Some(Some(clip_path)) = cx.cache.clip_path.get(current) {
385                canvas.clip_path(clip_path, ClipOp::Intersect, true);
386            }
387        }
388        canvas.set_matrix(&(transform.into()));
389
390        if let Some(view) = cx.views.remove(&current) {
391            view.draw(cx, canvas);
392            cx.views.insert(current, view);
393        }
394        canvas.restore();
395    }
396    let child_iter = DrawChildIterator::new(cx.tree, cx.current);
397    let descendants_are_backdrop_clipped = ancestor_backdrop_clip || backdrop_clip_count.is_some();
398    let child_space = ParentSpace::new(transform, clip_bounds(cx.cache, current), dirty_rect);
399
400    // Draw its children
401    for child in child_iter {
402        cx.current = child;
403        // TODO: Skip views with zero-sized bounding boxes here? Or let user decide if they want to skip?
404        draw_entity(
405            cx,
406            dirty_rect,
407            canvas,
408            current_z,
409            queue,
410            is_visible,
411            descendants_are_backdrop_clipped,
412            &child_space,
413        );
414    }
415
416    cx.current = current;
417
418    // Draw outline after children so it appears above descendant content.
419    if should_draw {
420        canvas.save();
421        if let Some(Some(clip_path)) = cx.cache.clip_path.get(current) {
422            canvas.clip_path(clip_path, ClipOp::Intersect, true);
423        }
424
425        canvas.set_matrix(&(transform.into()));
426
427        cx.draw_outline(canvas);
428        canvas.restore();
429    }
430
431    if let Some(count) = layer_count {
432        canvas.restore_to_count(count);
433    }
434    if let Some(count) = backdrop_clip_count {
435        canvas.restore_to_count(count);
436    }
437}

439fn is_non_empty(bounds: &BoundingBox) -> bool {
440    bounds.w > 0.0 && bounds.h > 0.0
441}
442
443fn intersects_non_empty(a: &BoundingBox, b: &BoundingBox) -> bool {
444    is_non_empty(a) && is_non_empty(b) && a.intersects(b)
445}
446
447fn union_non_empty(a: BoundingBox, b: BoundingBox) -> BoundingBox {
448    match (is_non_empty(&a), is_non_empty(&b)) {
449        (true, true) => a.union(&b),
450        (true, false) => a,
451        _ => b,
452    }
453}

455fn snap(rect: Rect) -> BoundingBox {
456    BoundingBox::from_min_max(
457        rect.left().floor(),
458        rect.top().floor(),
459        rect.right().ceil(),
460        rect.bottom().ceil(),
461    )
462}

464/// Maps bounds through a transform, expanding the result to whole pixels.
465fn map_bounds(transform: &Matrix, bounds: BoundingBox) -> BoundingBox {
466    if !is_non_empty(&bounds) {
467        return BoundingBox::default();
468    }
469
470    // Fast path for scale and translation, which avoids a call into Skia for every culled entity.
471    if is_scale_translate(transform) {
472        let (sx, sy) = (transform.scale_x(), transform.scale_y());
473        let (tx, ty) = (transform.translate_x(), transform.translate_y());
474        let (x0, x1) = (bounds.left() * sx + tx, bounds.right() * sx + tx);
475        let (y0, y1) = (bounds.top() * sy + ty, bounds.bottom() * sy + ty);
476        return snap(Rect::new(x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1)));
477    }
478
479    snap(transform.map_rect(Rect::from(bounds)).0)
480}

487/// Converts the cached draw bounds of an entity and its descendants into window coordinates, given
488/// the transform and clip bounds of the parent of the entity in window coordinates.
489pub(crate) fn subtree_to_screen(
490    bounds: &DrawBounds,
491    parent_transform: &Matrix,
492    parent_clip: Option<BoundingBox>,
493) -> BoundingBox {
494    let clipped = map_bounds(parent_transform, bounds.clipped);
495    let clipped = match parent_clip {
496        Some(clip) if is_non_empty(&clipped) => clipped.intersection(&clip),
497        _ => clipped,
498    };
499    if !is_non_empty(&bounds.escaped) {
500        return clipped;
501    }
502    union_non_empty(clipped, map_bounds(parent_transform, bounds.escaped))
503}

505/// Converts the cached bounds drawn by an entity itself into window coordinates.
506fn own_to_screen(
507    bounds: &DrawBounds,
508    parent_transform: &Matrix,
509    parent_clip: Option<BoundingBox>,
510) -> BoundingBox {
511    let own = map_bounds(parent_transform, bounds.own);
512    match parent_clip {
513        Some(clip) if !bounds.escapes_clip && is_non_empty(&own) => own.intersection(&clip),
514        _ => own,
515    }
516}

518/// Returns the transform and clip bounds, in window coordinates, which apply to the cached draw
519/// bounds of an entity.
520fn parent_space(
521    cache: &CachedData,
522    tree: &Tree<Entity>,
523    entity: Entity,
524) -> (Matrix, Option<BoundingBox>) {
525    if tree.is_window(entity) {
526        return (Matrix::new_identity(), None);
527    }
528
529    let parent = tree
530        .get_layout_parent(entity)
531        .unwrap_or_else(|| tree.get_parent_window(entity).unwrap_or(Entity::root()));
532
533    (cache.absolute_transform(tree, parent), clip_bounds(cache, parent))
534}

536/// Returns the previously computed draw bounds of an entity and its descendants in window
537/// coordinates, or `None` if they are not cached.
538pub(crate) fn cached_screen_bounds(
539    cache: &CachedData,
540    tree: &Tree<Entity>,
541    entity: Entity,
542) -> Option<BoundingBox> {
543    let bounds = cache.draw_bounds.get(entity)?;
544    let (parent_transform, parent_clip) = parent_space(cache, tree, entity);
545    Some(subtree_to_screen(bounds, &parent_transform, parent_clip))
546}

558/// Returns the draw bounds of an entity and its descendants in window coordinates, computing and

564    entity: Entity,

571/// Returns the cached draw bounds of an entity, computing and caching them (and those of its
572/// descendants) if necessary.
573fn cached_bounds(
574    style: &Style,
575    cache: &mut CachedData,
576    tree: &Tree<Entity>,
577    entity: Entity,
578) -> DrawBounds {
579    if let Some(bounds) = cache.draw_bounds.get(entity).copied() {
580        return bounds;
581    }
582
583    let bounds = draw_bounds(style, cache, tree, entity);
584    cache.draw_bounds.insert(entity, bounds);
585    bounds
586}

588/// Computes the draw bounds of an entity in the coordinate space of its parent.
589// Must be called after the layout and transform systems to be valid.
590fn draw_bounds(
591    style: &Style,
592    cache: &mut CachedData,
593    tree: &Tree<Entity>,
594    entity: Entity,
595) -> DrawBounds {
596    let mut own = decorated_bounds(style, cache, entity);
597
598    let mut clipped = BoundingBox::default();
599    let mut escaped = BoundingBox::default();
600    for child in DrawChildIterator::new(tree, entity) {
601        let child_bounds = cached_bounds(style, cache, tree, child);
602        clipped = union_non_empty(clipped, child_bounds.clipped);
603        escaped = union_non_empty(escaped, child_bounds.escaped);
604    }
605
606    if is_non_empty(&clipped) {
607        if let Some(clip) = own_clip_bounds(style, cache, tree, entity) {
608            clipped = clipped.intersection(&clip);
609        }
610    }
611    let mut clipped = union_non_empty(own, clipped);
612
613    if let Some(Filter::Blur(radius)) = style.filter.get(entity) {
614        let radius = radius.to_px().unwrap() * style.scale_factor();
615        own = own.expand(radius);
616        clipped = clipped.expand(radius);
617        if is_non_empty(&escaped) {
618            escaped = escaped.expand(radius);
619        }
620    }
621
622    let transform = cache.local_transform.get(entity).copied().unwrap_or_default();
623    let own = map_bounds(&transform, own);
624    let mut clipped = map_bounds(&transform, clipped);
625    let mut escaped = map_bounds(&transform, escaped);
626
627    // If z-index is not zero we can't be sure of the stacking context so we have to assume so and
628    // not clip.
629    let escapes_clip = tree.is_window(entity)
630        || !should_apply_entity_clip(style, entity)
631        || matches!(style.z_index.get(entity), Some(z_index) if *z_index != 0);
632
633    if escapes_clip {
634        escaped = union_non_empty(escaped, clipped);
635        clipped = BoundingBox::default();
636    }
637
638    DrawBounds { own, clipped, escaped, escapes_clip }
639}

641/// Returns the layout bounds of an entity expanded by its shadows and outline.
642fn decorated_bounds(style: &Style, cache: &CachedData, entity: Entity) -> BoundingBox {
643    let mut layout_bounds = cache.bounds.get(entity).copied().unwrap_or_default();
644    if !is_non_empty(&layout_bounds) {
645        return BoundingBox::default();
646    }
647
648    if let Some(shadows) = style.shadow.get_resolved(entity, &style.custom_shadow_props) {
649        let original_bounds = layout_bounds;
650
651        for shadow in shadows.iter().filter(|shadow| !shadow.inset) {
652            let mut shadow_bounds = original_bounds;
653
654            let scale_factor = style.scale_factor();
655
656            let x = shadow.x_offset.to_px().unwrap() * scale_factor;
657            let y = shadow.y_offset.to_px().unwrap() * scale_factor;
658
659            shadow_bounds = shadow_bounds.offset(x, y);
660
661            if let Some(blur_radius) = shadow.blur_radius.as_ref().and_then(Length::to_px) {
662                shadow_bounds = shadow_bounds.expand(blur_radius * scale_factor);
663            }
664
665            if let Some(spread_radius) = shadow.spread_radius.as_ref().and_then(Length::to_px) {
666                shadow_bounds = shadow_bounds.expand(spread_radius * scale_factor);
667            }
668
669            layout_bounds = layout_bounds.union(&shadow_bounds);
670        }
671    }
672
673    let mut outline_bounds = layout_bounds;
674
675    if let Some(outline_width) =
676        style.outline_width.get_resolved(entity, &style.custom_length_props)
677    {
678        outline_bounds = outline_bounds
679            .expand(outline_width.to_pixels(layout_bounds.diagonal(), style.scale_factor()));
680    }
681
682    if let Some(outline_offset) =
683        style.outline_offset.get_resolved(entity, &style.custom_length_props)
684    {
685        outline_bounds = outline_bounds
686            .expand(outline_offset.to_pixels(layout_bounds.diagonal(), style.scale_factor()));
687    }
688
689    layout_bounds.union(&outline_bounds)
690}

692/// The coordinate space of the parent of an entity being drawn.
693#[derive(Clone, Copy)]
694struct ParentSpace {
695    /// The transform of the parent in window coordinates.
696    transform: Matrix,
697    /// The clip bounds of the parent in window coordinates.
698    clip: Option<BoundingBox>,
699    /// The dirty rect in the coordinate space of the parent, if the parent transform only scales
700    /// and translates. Allows children to be culled using their cached draw bounds directly.
701    local_dirty: Option<LocalDirty>,
702}
703
704#[derive(Clone, Copy)]
705struct LocalDirty {
706    /// The dirty rect intersected with the parent clip, for comparison with clipped bounds.
707    clipped: BoundingBox,
708    /// The dirty rect, for comparison with bounds which escape the parent clip.
709    escaped: BoundingBox,
710}
711
712impl ParentSpace {
713    fn new(transform: Matrix, clip: Option<BoundingBox>, dirty_rect: &Option<BoundingBox>) -> Self {
714        let local_dirty = dirty_rect.and_then(|dirty| {
715            let (sx, sy) = (transform.scale_x(), transform.scale_y());
716            if !is_scale_translate(&transform) || sx == 0.0 || sy == 0.0 {
717                return None;
718            }
719            let (tx, ty) = (transform.translate_x(), transform.translate_y());
720            let to_local = |bounds: BoundingBox| {
721                if !is_non_empty(&bounds) {
722                    return BoundingBox::default();
723                }
724                // Window space bounds are expanded to whole pixels, so expand the dirty rect to
725                // keep culling conservative.
726                let bounds = bounds.expand(1.0);
727                let (x0, x1) = ((bounds.left() - tx) / sx, (bounds.right() - tx) / sx);
728                let (y0, y1) = ((bounds.top() - ty) / sy, (bounds.bottom() - ty) / sy);
729                BoundingBox::from_min_max(x0.min(x1), y0.min(y1), x0.max(x1), y0.max(y1))
730            };
731            let clipped = clip.map_or(dirty, |clip| dirty.intersection(&clip));
732            Some(LocalDirty { clipped: to_local(clipped), escaped: to_local(dirty) })
733        });
734        Self { transform, clip, local_dirty }
735    }
736
737    /// Returns true if the drawn bounds of an entity and its descendants may intersect the dirty
738    /// rect.
739    fn intersects_dirty(&self, bounds: &DrawBounds, dirty_rect: &BoundingBox) -> bool {
740        match &self.local_dirty {
741            Some(local) => {
742                intersects_non_empty(&bounds.clipped, &local.clipped)
743                    || intersects_non_empty(&bounds.escaped, &local.escaped)
744            }
745            None => intersects_non_empty(
746                &subtree_to_screen(bounds, &self.transform, self.clip),
747                dirty_rect,
748            ),
749        }
750    }
751}

903    /// Styles a `.probe` view, a child of a `.parent` view, with `css`, paints the probe's shadows
904    /// and background with its bounds at (10, 10, 20, 20) on a 60x40 surface, and returns a lookup

907        use skia_safe::{AlphaType, ColorType, ImageInfo};

909        let mut cx = Context::default();

911        let parent = Element::new(&mut cx).class("parent").entity();
912        let probe = cx.with_current(parent, |cx| Element::new(cx).class("probe").entity());
913        crate::systems::style_system(&mut cx);
914        cx.cache.set_bounds(probe, BoundingBox { x: 10.0, y: 10.0, w: 20.0, h: 20.0 });
915
916        let info = ImageInfo::new((60, 40), ColorType::RGBA8888, AlphaType::Premul, None);
917        let mut surface = skia_safe::surfaces::raster(&info, None, None).unwrap();
918        let mut draw_cx = DrawContext {
919            current: probe,
920            style: &cx.style,
921            cache: &mut cx.cache,
922            tree: &cx.tree,
923            models: &cx.models,
924            views: &mut cx.views,
925            resource_manager: &cx.resource_manager,
926            text_context: &mut cx.text_context,
927            modifiers: &cx.modifiers,
928            mouse: &cx.mouse,
929            windows: &mut cx.windows,
930        };
931        draw_cx.draw_shadows(surface.canvas());
932        draw_cx.draw_background(surface.canvas());
933
934        let mut pixels = vec![0u8; 60 * 40 * 4];
935        assert!(surface.read_pixels(&info, &mut pixels, 60 * 4, (0, 0)));

939        }
940    }
