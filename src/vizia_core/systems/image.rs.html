7// Iterate the tree and load any images used by entities which aren't already loaded.
8// Remove any images no longer being used.
9pub(crate) fn image_system(cx: &mut Context) {
10    let cx = &mut ResourceContext::new(cx);
11
12    cx.resource_manager.mark_images_unused();
13
14    // Iterate the tree and load any defined images that aren't already loaded
15    for entity in cx.tree.into_iter() {
16        // Load a background-image if the entity has one
17        if let Some(background_images) = cx.style.background_image.get(entity).cloned() {
18            for image in background_images.iter() {
19                match image {
20                    ImageOrGradient::Image(name) => {
21                        associate_image(cx, entity, name);
22                    }
23                    _ => {}
24                }
25            }
26        }
27    }
28
29    cx.resource_manager.evict_unused_images();
30}
31
32fn associate_image(cx: &mut ResourceContext, entity: Entity, image_name: &str) -> bool {
33    if let Some(image_id) = cx.resource_manager.image_ids.get(image_name) {
34        // Check if the image is already loaded
35        if let Some(image_store) = cx.resource_manager.images.get_mut(image_id) {
36            image_store.observers.insert(entity);
37            image_store.used = true;
38
39            return true;
40        }
41    }
42
43    false
44}
