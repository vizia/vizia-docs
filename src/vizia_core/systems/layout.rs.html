13fn hover_geometry_changed(entity: Entity, hovered: Entity, geo: GeoChanged) -> bool {
14    entity == hovered
15        && geo.intersects(
16            GeoChanged::POSX_CHANGED
17                | GeoChanged::POSY_CHANGED
18                | GeoChanged::WIDTH_CHANGED
19                | GeoChanged::HEIGHT_CHANGED,
20        )
21}

23/// Returns whether an entity is a valid restart point for incremental layout, mirroring
24/// morphorm's `NodeExt::is_restartable`: its width and height must both be `Pixels` or `Stretch`.
25fn is_restartable(style: &Style, entity: Entity) -> bool {
26    let width =
27        style.width.get_resolved(entity, &style.custom_units_props).unwrap_or(Units::Stretch(1.0));
28    let height =
29        style.height.get_resolved(entity, &style.custom_units_props).unwrap_or(Units::Stretch(1.0));
30    matches!(width, Units::Pixels(_) | Units::Stretch(_))
31        && matches!(height, Units::Pixels(_) | Units::Stretch(_))
32}

34/// Returns the ancestor that morphorm restarts layout from for a dirty `entity`, mirroring
35/// `NodeExt::find_relayout_root`. Used to determine which views undergo layout for the debug overlay.
36fn layout_restart_root(style: &Style, tree: &Tree<Entity>, entity: Entity) -> Entity {
37    let mut root = match tree.get_layout_parent(entity) {
38        Some(parent) => parent,
39        None => return entity,
40    };
41    // If the dirty node is itself absolutely positioned it is out of its parent's flow, so a change
42    // to it cannot alter the parent's size — restart from the parent without walking up (mirrors
43    // morphorm).
44    if style.position_type.get(entity).copied().unwrap_or_default()
45        == morphorm::PositionType::Absolute
46    {
47        return root;
48    }
49    while let Some(parent) = tree.get_layout_parent(root) {
50        // An absolutely-positioned node is out of its parent's flow, so its size never affects the
51        // parent's layout — stop here regardless of its own sizing (mirrors morphorm).
52        if style.position_type.get(root).copied().unwrap_or_default()
53            == morphorm::PositionType::Absolute
54        {
55            break;
56        }
57        if is_restartable(style, root) {
58            break;
59        }
60        root = parent;
61    }
62    root
63}

65/// Determines the size and position of views.
66///
67/// Relayout is performed incrementally: only the subtrees rooted at the dirty nodes (tracked in
68/// [`Style::relayout`](crate::style::Style)) are recomputed. Dirty descendants covered by a dirty
69/// ancestor are collapsed away via [`minimal_layout_dirty_roots`], and for each remaining root
70/// `Node::layout` walks up to the best ancestor to restart layout from before recomputing that
71/// subtree. Marking the root dirty performs a full relayout.
72pub(crate) fn layout_system(cx: &mut Context) {
73    let _scope = crate::profiling::scope("layout_system");
74    let mut hover_geometry_affected = false;
75    text_system(cx);
76
77    if !cx.style.relayout.is_empty() {
78        // Determine the minimal set of subtree roots to lay out. Marking the root dirty triggers a full relayout.
79        // Otherwise, relayout only the minimal set of dirty subtrees.
80        let relayout_roots: Vec<Entity> = if cx.style.relayout.contains(&Entity::root()) {
81            vec![Entity::root()]
82        } else {
83            let dirty = std::mem::take(&mut cx.style.relayout);
84            minimal_layout_dirty_roots(&cx.tree, &dirty).into_iter().collect()
85        };
86        cx.style.relayout.clear();
87
88        // Debug overlay: record which views undergo layout (the subtree of each restart root).
89        // The overlay persists until the next layout pass, so schedule a redraw of the previously
90        // highlighted views to erase their outlines before recording the new set.
91        if cx.style.debug_layout {
92            for entity in std::mem::take(&mut cx.style.laid_out) {
93                cx.needs_redraw(entity);
94            }
95            let mut laid_out = HashSet::new();
96            for &root in &relayout_roots {
97                let restart = layout_restart_root(&cx.style, &cx.tree, root);
98                for entity in LayoutTreeIterator::subtree(&cx.tree, restart) {
99                    laid_out.insert(entity);
100                }
101            }
102            cx.style.laid_out = laid_out;
103        }
104
105        for entity in relayout_roots {
106            entity.layout(
107                &mut cx.cache,
108                &cx.tree,
109                &cx.style,
110                &mut SubLayout {
111                    text_context: &mut cx.text_context,
112                    resource_manager: &cx.resource_manager,
113                },
114            );
115        }
116
117        let cx = &mut EventContext::new(cx);
118
119        let iter = LayoutTreeIterator::full(cx.tree);
120
121        for entity in iter {
122            cx.current = entity;
123            if cx.style.display.get(entity).copied().unwrap_or_default() == Display::None {
124                continue;
125            }
126            // Morphorm produces relative positions so convert to absolute.
127            if let Some(parent) = cx.tree.get_layout_parent(entity) {
128                let parent_bounds = cx.cache.get_bounds(parent);
129                if let Some(bounds) = cx.cache.bounds.get_mut(entity) {
130                    if let Some(relative_bounds) = cx.cache.relative_bounds.get(entity) {
131                        let x = relative_bounds.x + parent_bounds.x;
132                        let y = relative_bounds.y + parent_bounds.y;
133                        let w = relative_bounds.w;
134                        let h = relative_bounds.h;
135
136                        let mut geo_changed = GeoChanged::empty();
137
138                        if x != bounds.x {
139                            geo_changed.set(GeoChanged::POSX_CHANGED, true);
140                        }
141
142                        if y != bounds.y {
143                            geo_changed.set(GeoChanged::POSY_CHANGED, true);
144                        }
145
146                        if w != bounds.w {
147                            geo_changed.set(GeoChanged::WIDTH_CHANGED, true);
148                            cx.cache.path.remove(entity);
149                        }
150
151                        if h != bounds.h {
152                            geo_changed.set(GeoChanged::HEIGHT_CHANGED, true);
153                            cx.cache.path.remove(entity);
154                        }
155
156                        if let Some(geo) = cx.cache.geo_changed.get_mut(entity) {
157                            *geo = geo_changed;
158                        }
159
160                        let new_bounds = BoundingBox { x, y, w, h };
161
162                        // if new_bounds != *bounds && *bounds != BoundingBox::default() {
163                        //     cx.needs_redraw();
164                        // }
165
166                        *bounds = new_bounds;
167                    }
168                }
169            }
170
171            if let Some(geo) = cx.cache.geo_changed.get(entity).copied() {
172                hover_geometry_affected |= hover_geometry_changed(entity, *cx.hovered, geo);
173
174                if !geo.is_empty() && cx.style.text.contains(entity) {
175                    cx.style.needs_text_layout(entity);
176                }
177
178                if !geo.is_empty()
179                // && cx.style.text.get(entity).is_some()
180                {
181                    cx.needs_redraw();
182
183                    cx.needs_retransform();
184                    cx.needs_reclip();
185
186                    // Cached draw bounds are independent of ancestor clips, except that clips with
187                    // a single hidden axis extend to the window bounds.
188                    if cx.tree.is_window(entity) {
189                        for descendant in LayoutTreeIterator::subtree(cx.tree, entity).skip(1) {
190                            cx.cache.draw_bounds.remove(descendant);
191                        }
192                    }
193                }
194
195                // TODO: Use geo changed to determine whether an entity needs to be redrawn.
196
197                if !geo.is_empty() {
198                    let mut event = Event::new(WindowEvent::GeometryChanged(geo))
199                        .target(entity)
200                        .origin(entity)
201                        .propagate(Propagation::Direct);
202                    visit_entity(cx, entity, &mut event);
203                }
204            }
205
206            if let Some(geo) = cx.cache.geo_changed.get_mut(entity) {
207                *geo = GeoChanged::empty();
208            }
209        }
210
211        cx.style.system_flags.set(SystemFlags::RELAYOUT, false);
212    }
213
214    text_layout_system(cx);
215    let transform_hover_changed = transform_system(cx);
216    let clipping_hover_changed = clipping_system(cx);
217
218    if (hover_geometry_affected || transform_hover_changed || clipping_hover_changed)
219        && cx.captured.is_null()
220    {
221        if let Some(proxy) = &cx.event_proxy {
222            let event = ProxyEvent::new(InternalEvent::RefreshHover)
223                .target(Entity::root())
224                .origin(Entity::root())
225                .propagate(Propagation::Up);
226
227            proxy.send(event).expect("Failed to send event");
228        }
229    }
230}

232fn visit_entity(cx: &mut EventContext, entity: Entity, event: &mut Event) {
233    // Send event to models attached to the entity
234    if let Some(ids) =
235        cx.models.get(&entity).map(|models| models.keys().cloned().collect::<Vec<_>>())
236    {
237        for id in ids {
238            if let Some(mut model) =
239                cx.models.get_mut(&entity).and_then(|models| models.remove(&id))
240            {
241                cx.current = entity;
242
243                model.event(cx, event);
244
245                cx.models.get_mut(&entity).and_then(|models| models.insert(id, model));
246            }
247        }
248    }
249
250    // Return early if the event was consumed by a model
251    if event.meta.consumed {
252        return;
253    }
254
255    // Send event to the view attached to the entity
256    if let Some(mut view) = cx.views.remove(&entity) {
257        cx.current = entity;
258        view.event(cx, event);
259
260        cx.views.insert(entity, view);
261    }
262}
