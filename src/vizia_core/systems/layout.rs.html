12/// Returns whether an entity is a valid restart point for incremental layout, mirroring
13/// morphorm's `NodeExt::is_restartable`: its width and height must both be `Pixels` or `Stretch`.
14fn is_restartable(style: &Style, entity: Entity) -> bool {
15    let width =
16        style.width.get_resolved(entity, &style.custom_units_props).unwrap_or(Units::Stretch(1.0));
17    let height =
18        style.height.get_resolved(entity, &style.custom_units_props).unwrap_or(Units::Stretch(1.0));
19    matches!(width, Units::Pixels(_) | Units::Stretch(_))
20        && matches!(height, Units::Pixels(_) | Units::Stretch(_))
21}

23/// Returns the ancestor that morphorm restarts layout from for a dirty `entity`, mirroring
24/// `NodeExt::find_relayout_root`. Used to determine which views undergo layout for the debug overlay.
25fn layout_restart_root(style: &Style, tree: &Tree<Entity>, entity: Entity) -> Entity {
26    let mut root = match tree.get_layout_parent(entity) {
27        Some(parent) => parent,
28        None => return entity,
29    };
30    // If the dirty node is itself absolutely positioned it is out of its parent's flow, so a change
31    // to it cannot alter the parent's size — restart from the parent without walking up (mirrors
32    // morphorm).
33    if style.position_type.get(entity).copied().unwrap_or_default()
34        == morphorm::PositionType::Absolute
35    {
36        return root;
37    }
38    while let Some(parent) = tree.get_layout_parent(root) {
39        // An absolutely-positioned node is out of its parent's flow, so its size never affects the
40        // parent's layout — stop here regardless of its own sizing (mirrors morphorm).
41        if style.position_type.get(root).copied().unwrap_or_default()
42            == morphorm::PositionType::Absolute
43        {
44            break;
45        }
46        if is_restartable(style, root) {
47            break;
48        }
49        root = parent;
50    }
51    root
52}

54/// Determines the size and position of views.
55///
56/// Relayout is performed incrementally: only the subtrees rooted at the dirty nodes (tracked in
57/// [`Style::relayout`](crate::style::Style)) are recomputed. Dirty descendants covered by a dirty
58/// ancestor are collapsed away via [`minimal_layout_dirty_roots`], and for each remaining root
59/// `Node::layout` walks up to the best ancestor to restart layout from before recomputing that
60/// subtree. Marking the root dirty performs a full relayout.
61pub(crate) fn layout_system(cx: &mut Context) {
62    text_system(cx);
63
64    if !cx.style.relayout.is_empty() {
65        // Determine the minimal set of subtree roots to lay out. Marking the root dirty triggers a full relayout.
66        // Otherwise, relayout only the minimal set of dirty subtrees.
67        let relayout_roots: Vec<Entity> = if cx.style.relayout.contains(&Entity::root()) {
68            vec![Entity::root()]
69        } else {
70            let dirty = std::mem::take(&mut cx.style.relayout);
71            minimal_layout_dirty_roots(&cx.tree, &dirty).into_iter().collect()
72        };
73        cx.style.relayout.clear();
74
75        // Debug overlay: record which views undergo layout (the subtree of each restart root).
76        // The overlay persists until the next layout pass, so schedule a redraw of the previously
77        // highlighted views to erase their outlines before recording the new set.
78        if cx.style.debug_layout {
79            for entity in std::mem::take(&mut cx.style.laid_out) {
80                cx.needs_redraw(entity);
81            }
82            let mut laid_out = HashSet::new();
83            for &root in &relayout_roots {
84                let restart = layout_restart_root(&cx.style, &cx.tree, root);
85                for entity in LayoutTreeIterator::subtree(&cx.tree, restart) {
86                    laid_out.insert(entity);
87                }
88            }
89            cx.style.laid_out = laid_out;
90        }
91
92        for entity in relayout_roots {
93            entity.layout(
94                &mut cx.cache,
95                &cx.tree,
96                &cx.style,
97                &mut SubLayout {
98                    text_context: &mut cx.text_context,
99                    resource_manager: &cx.resource_manager,
100                },
101            );
102        }
103
104        let cx = &mut EventContext::new(cx);
105
106        let iter = LayoutTreeIterator::full(cx.tree);
107
108        for entity in iter {
109            cx.current = entity;
110            if cx.style.display.get(entity).copied().unwrap_or_default() == Display::None {
111                continue;
112            }
113            // Morphorm produces relative positions so convert to absolute.
114            if let Some(parent) = cx.tree.get_layout_parent(entity) {
115                let parent_bounds = cx.cache.get_bounds(parent);
116                if let Some(bounds) = cx.cache.bounds.get_mut(entity) {
117                    if let Some(relative_bounds) = cx.cache.relative_bounds.get(entity) {
118                        let x = relative_bounds.x + parent_bounds.x;
119                        let y = relative_bounds.y + parent_bounds.y;
120                        let w = relative_bounds.w;
121                        let h = relative_bounds.h;
122
123                        let mut geo_changed = GeoChanged::empty();
124
125                        if x != bounds.x {
126                            geo_changed.set(GeoChanged::POSX_CHANGED, true);
127                        }
128
129                        if y != bounds.y {
130                            geo_changed.set(GeoChanged::POSY_CHANGED, true);
131                        }
132
133                        if w != bounds.w {
134                            geo_changed.set(GeoChanged::WIDTH_CHANGED, true);
135                            cx.cache.path.remove(entity);
136                        }
137
138                        if h != bounds.h {
139                            geo_changed.set(GeoChanged::HEIGHT_CHANGED, true);
140                            cx.cache.path.remove(entity);
141                        }
142
143                        if let Some(geo) = cx.cache.geo_changed.get_mut(entity) {
144                            *geo = geo_changed;
145                        }
146
147                        let new_bounds = BoundingBox { x, y, w, h };
148
149                        // if new_bounds != *bounds && *bounds != BoundingBox::default() {
150                        //     cx.needs_redraw();
151                        // }
152
153                        *bounds = new_bounds;
154                    }
155                }
156            }
157
158            if let Some(geo) = cx.cache.geo_changed.get(entity).copied() {
159                if !geo.is_empty() && cx.style.text.contains(entity) {
160                    cx.style.needs_text_layout(entity);
161                }
162
163                if !geo.is_empty()
164                // && cx.style.text.get(entity).is_some()
165                {
166                    cx.needs_redraw();
167
168                    cx.needs_retransform();
169                    cx.needs_reclip();
170
171                    // If the entity clips its children (Overflow::Hidden or ClipPath::Shape)
172                    // and its geometry changed, the clip path has changed too, so invalidate
173                    // all descendants' cached draw_bounds.
174                    if matches!(cx.style.overflowx.get(entity), Some(Overflow::Hidden))
175                        || matches!(cx.style.overflowy.get(entity), Some(Overflow::Hidden))
176                        || matches!(cx.style.clip_path.get(entity), Some(ClipPath::Shape(_)))
177                    {
178                        for descendant in LayoutTreeIterator::subtree(cx.tree, entity).skip(1) {
179                            cx.cache.draw_bounds.remove(descendant);
180                        }
181                    }
182                }
183
184                // TODO: Use geo changed to determine whether an entity needs to be redrawn.
185
186                if !geo.is_empty() {
187                    let mut event = Event::new(WindowEvent::GeometryChanged(geo))
188                        .target(entity)
189                        .origin(entity)
190                        .propagate(Propagation::Direct);
191                    visit_entity(cx, entity, &mut event);
192                }
193            }
194
195            if let Some(geo) = cx.cache.geo_changed.get_mut(entity) {
196                *geo = GeoChanged::empty();
197            }
198        }
199
200        // A relayout, retransform, or reclip, can cause the element under the cursor to change. So we push a mouse move event here to force
201        // a new event cycle and the hover system to trigger.
202        // However, skip this during capture to avoid sending stale coordinates that interfere with drag operations.
203        if let Some(proxy) = &cx.event_proxy {
204            if cx.captured.is_null() {
205                let event =
206                    ProxyEvent::new(WindowEvent::MouseMove(cx.mouse.cursor_x, cx.mouse.cursor_y))
207                        .target(Entity::root())
208                        .origin(Entity::root())
209                        .propagate(Propagation::Up);
210
211                proxy.send(event).expect("Failed to send event");
212            }
213        }
214
215        cx.style.system_flags.set(SystemFlags::RELAYOUT, false);
216    }
217
218    text_layout_system(cx);
219    transform_system(cx);
220    clipping_system(cx);
221}

223fn visit_entity(cx: &mut EventContext, entity: Entity, event: &mut Event) {
224    // Send event to models attached to the entity
225    if let Some(ids) =
226        cx.models.get(&entity).map(|models| models.keys().cloned().collect::<Vec<_>>())
227    {
228        for id in ids {
229            if let Some(mut model) =
230                cx.models.get_mut(&entity).and_then(|models| models.remove(&id))
231            {
232                cx.current = entity;
233
234                model.event(cx, event);
235
236                cx.models.get_mut(&entity).and_then(|models| models.insert(id, model));
237            }
238        }
239    }
240
241    // Return early if the event was consumed by a model
242    if event.meta.consumed {
243        return;
244    }
245
246    // Send event to the view attached to the entity
247    if let Some(mut view) = cx.views.remove(&entity) {
248        cx.current = entity;
249        view.event(cx, event);
250
251        cx.views.insert(entity, view);
252    }
253}
