1use vizia_storage::LayoutTreeIterator;
2
3use crate::{animation::Interpolator, prelude::*};

5/// Applies transforms to the layout tree.
6pub(crate) fn transform_system(cx: &mut Context) {
7    if cx.style.retransform.is_empty() {
8        return;
9    }
10
11    let iter = LayoutTreeIterator::full(&cx.tree);
12
13    for entity in iter {
14        if !cx.style.retransform.contains(&entity) {
15            continue;
16        }
17
18        let bounds = cx.cache.bounds.get(entity).copied().unwrap();
19        if let Some(parent) = cx.tree.get_layout_parent(entity) {
20            let parent_transform = cx.cache.transform.get(parent).copied().unwrap();
21            if let Some(tx) = cx.cache.transform.get_mut(entity) {
22                let scale_factor = cx.style.scale_factor();
23
24                // Apply transform origin.
25                let mut origin = cx
26                    .style
27                    .transform_origin
28                    .get(entity)
29                    .map(|transform_origin| {
30                        let mut origin = skia_safe::Matrix::translate(bounds.top_left());
31                        let offset = transform_origin.as_transform(bounds, scale_factor);
32                        origin = offset * origin;
33                        origin
34                    })
35                    .unwrap_or(skia_safe::Matrix::translate(bounds.center()));
36                // transform = origin * transform;
37                let mut transform = origin;
38                origin = origin.invert().unwrap();
39
40                // Apply translation.
41                if let Some(translate) = cx.style.translate.get(entity) {
42                    transform = transform * translate.as_transform(bounds, scale_factor);
43                }
44
45                // Apply rotation.
46                if let Some(rotate) = cx.style.rotate.get(entity) {
47                    transform = transform * rotate.as_transform(bounds, scale_factor);
48                }
49
50                // Apply scaling.
51                if let Some(scale) = cx.style.scale.get(entity) {
52                    transform = transform * scale.as_transform(bounds, scale_factor);
53                }
54
55                // Apply transform functions.
56                if let Some(transforms) = cx.style.transform.get(entity) {
57                    // Check if the transform is currently animating
58                    // Get the animation state
59                    // Manually interpolate the value to get the overall transform for the current frame
60                    if let Some(animation_state) = cx.style.transform.get_active_animation(entity) {
61                        if let Some(start) = animation_state.keyframes.first() {
62                            if let Some(end) = animation_state.keyframes.last() {
63                                let start_transform =
64                                    start.value.as_transform(bounds, scale_factor);
65                                let end_transform = end.value.as_transform(bounds, scale_factor);
66                                let t = animation_state.t;
67                                let animated_transform = skia_safe::Matrix::interpolate(
68                                    &start_transform,
69                                    &end_transform,
70                                    t,
71                                );
72                                transform = transform * animated_transform;
73                            }
74                        }
75                    } else {
76                        transform = transform * transforms.as_transform(bounds, scale_factor);
77                    }
78                }
79
80                transform = transform * origin;
81
82                let new_transform = parent_transform * transform;
83
84                if *tx != new_transform {
85                    cx.style.needs_reclip(entity);
86                    let iter = LayoutTreeIterator::subtree(&cx.tree, entity);
87                    for descendant in iter {
88                        cx.style.needs_reclip(descendant);
89                        // Cached draw bounds include transformed and clipped extents.
90                        // Invalidate them when ancestor transforms change so dirty/intersection
91                        // checks don't cull newly visible descendants.
92                        if descendant != entity {
93                            cx.cache.draw_bounds.remove(descendant);
94                        }
95                    }
96                }
97
98                *tx = new_transform;
99            }
100        }
101    }
102
103    cx.style.retransform.clear();
104}
