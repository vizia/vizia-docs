3/// A direction for resizing a resizable stack, either horizontally (right) or vertically (bottom).
4#[derive(PartialEq, Clone, Copy)]
5pub enum ResizeStackDirection {
6    Right,
7    Bottom,
8}
9
10/// A view that can be resized by clicking and dragging from the right or bottom edge.
11///
12/// The `ResizableStack` struct allows users to create a resizable container in a user interface.
13/// It supports resizing in either a horizontal (right) or vertical (bottom) direction, as specified
14/// by the `direction` field. The resizing behavior is controlled via the `on_drag` callback, which
15/// is triggered during a drag operation.
16#[derive(Lens)]
17pub struct ResizableStack {
18    /// Tracks whether the edge of the view is currently being dragged.
19    is_dragging: bool,
20
21    /// A callback function that is triggered when the view is being dragged.
22    /// The callback receives a mutable reference to the event context and the new size
23    /// of the stack as a floating-point value.
24    on_drag: Box<dyn Fn(&mut EventContext, f32)>,
25
26    /// Specifies the direction in which the stack can be resized.
27    /// This can be either `Right` for horizontal resizing or `Bottom` for vertical resizing.
28    direction: ResizeStackDirection,
29}
30
31impl ResizableStack {
32    /// Creates a new `ResizableStack` view.
33    /// The `size` parameter is a lens to the size of the stack, which will be updated when the stack is resized.
34    /// The `direction` parameter specifies whether the stack is resized horizontally (right) or vertically (bottom).
35    /// The `on_drag` callback is called with the new size when the stack is being resized.
36    /// The `content` closure is called to build the content of the stack.
37    pub fn new<F>(
38        cx: &mut Context,
39        size: impl Lens<Target = Units>,
40        direction: ResizeStackDirection,
41        on_drag: impl Fn(&mut EventContext, f32) + 'static,
42        content: F,
43    ) -> Handle<Self>
44    where
45        F: FnOnce(&mut Context),
46    {
47        let handle = Self { is_dragging: false, on_drag: Box::new(on_drag), direction }
48            .build(cx, |cx| {
49                Element::new(cx)
50                    .position_type(PositionType::Absolute)
51                    .z_index(10)
52                    .class("resize-handle")
53                    .on_press_down(|cx| cx.emit(ResizableStackEvent::StartDrag));
54
55                (content)(cx);
56            })
57            .toggle_class(
58                "horizontal",
59                ResizableStack::direction.map(|d| *d == ResizeStackDirection::Bottom),
60            )
61            .toggle_class(
62                "vertical",
63                ResizableStack::direction.map(|d| *d == ResizeStackDirection::Right),
64            );
65
66        if direction == ResizeStackDirection::Right {
67            handle.width(size)
68        } else {
69            handle.height(size)
70        }
71    }
72}
73
74/// Events emitted by the `ResizableStack` view to indicate changes in dragging state.
75pub enum ResizableStackEvent {
76    /// Emitted when the user starts dragging the resizable edge of the stack.
77    /// This event is triggered when the user presses down on the resize handle.
78    /// It enables dragging behavior and locks the cursor.
79    StartDrag,
80
81    /// Emitted when the user stops dragging the resizable edge of the stack.
82    /// This event is triggered when the user releases the mouse button after dragging.
83    /// It disables dragging behavior and unlocks the cursor.
84    StopDrag,
85}
86
87impl View for ResizableStack {
88    fn element(&self) -> Option<&'static str> {
89        Some("resizable-stack")
90    }
91
92    fn event(&mut self, cx: &mut EventContext, event: &mut Event) {
93        event.map(|resizable_stack_event, event| match resizable_stack_event {
94            ResizableStackEvent::StartDrag => {
95                self.is_dragging = true;
96                cx.set_active(true);
97                cx.capture();
98                cx.lock_cursor_icon();
99
100                // Disable pointer events for everything while dragging
101                cx.with_current(Entity::root(), |cx| {
102                    cx.set_pointer_events(false);
103                });
104
105                // Prevent propagation in case the resizable stack is within another resizable stack
106                event.consume();
107            }
108
109            ResizableStackEvent::StopDrag => {
110                self.is_dragging = false;
111                cx.set_active(false);
112                cx.release();
113                cx.unlock_cursor_icon();
114
115                // Re-enable pointer events
116                cx.with_current(Entity::root(), |cx| {
117                    cx.set_pointer_events(true);
118                });
119
120                event.consume()
121            }
122        });
123
124        event.map(|window_event, _| match window_event {
125            WindowEvent::MouseMove(x, y) => {
126                let dpi = cx.scale_factor();
127                if self.is_dragging {
128                    let new_size = if self.direction == ResizeStackDirection::Right {
129                        let posx = cx.bounds().x;
130                        (*x - posx) / dpi
131                    } else {
132                        let posy = cx.bounds().y;
133                        (*y - posy) / dpi
134                    };
135
136                    if new_size.is_finite() && new_size > 5.0 {
137                        (self.on_drag)(cx, new_size);
138                    }
139                }
140            }
141
142            WindowEvent::MouseUp(button) if *button == MouseButton::Left => {
143                cx.emit(ResizableStackEvent::StopDrag);
144            }
145
146            _ => {}
147        });
148    }
149}
